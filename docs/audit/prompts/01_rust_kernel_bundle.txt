```rs
//! forwards input; the game is whatever Lua scripts the data directory holds.
//!
//!   auto [GAME_DIR] [--headless TICKS] [--screenshot OUT.ppm] [--frames N] [--size WxH] [--save-dir DIR]

struct Opts {
    root: PathBuf,
    headless: Option<u64>,
    screenshot: Option<String>,
    frames: Option<u64>,
    ticks_per_frame: u32,
    size: (i32, i32),
    save_dir: Option<PathBuf>,
    fps: u32,
    fullscreen: bool,
}

fn parse_args() -> Result<Opts, String> {
    let mut o = Opts {
        root: PathBuf::from("game"),
        headless: None,
        screenshot: None,
        frames: None,
        ticks_per_frame: 1,
        size: (1280, 720),
        save_dir: None,
        fps: 60,
        fullscreen: false,
    };
    let mut args = std::env::args().skip(1);
    while let Some(a) = args.next() {
        let mut val = |name: &str| args.next().ok_or(format!("{name} needs a value"));
        match a.as_str() {
            "--headless" => o.headless = Some(val("--headless")?.parse().map_err(|_| "bad tick count")?),
            "--screenshot" => o.screenshot = Some(val("--screenshot")?),
            "--frames" => o.frames = Some(val("--frames")?.parse().map_err(|_| "bad frame count")?),
            "--ticks-per-frame" => o.ticks_per_frame = val("--ticks-per-frame")?.parse().map_err(|_| "bad tick count")?,
            "--save-dir" => o.save_dir = Some(PathBuf::from(val("--save-dir")?)),
            "--fps" => o.fps = val("--fps")?.parse().map_err(|_| "bad fps")?,
            "--fullscreen" => o.fullscreen = true,
            "--size" => {
                let v = val("--size")?;
                let (w, h) = v.split_once('x').ok_or("size is WxH")?;
                o.size = (w.parse().map_err(|_| "bad width")?, h.parse().map_err(|_| "bad height")?);
            }
            "-V" | "--version" => {
                println!("auto {}", env!("CARGO_PKG_VERSION"));
                std::process::exit(0);
            }
            "-h" | "--help" => {
                println!(
                    "usage: auto [GAME_DIR] [--headless TICKS] [--screenshot OUT.ppm] [--frames N] [--ticks-per-frame K] [--size WxH] [--save-dir DIR] [--fps N] [--fullscreen]"
                );
                std::process::exit(0);
            }
            s if s.starts_with('-') => return Err(format!("unknown flag {s}")),
            s => o.root = PathBuf::from(s),
        }
    }
    // Released builds ship `game/` next to the executable; find it from anywhere.
    if !o.root.join("main.lua").exists()
        && o.root == Path::new("game")
        && let Some(beside) = std::env::current_exe().ok().and_then(|p| Some(p.parent()?.join("game")))
    {
        o.root = beside;
    }
    if !o.root.join("main.lua").exists() {
        return Err(format!("no main.lua in {}", o.root.display()));
    }
    o.root = plain_path(o.root.canonicalize().map_err(|e| e.to_string())?);
    if let Some(d) = &o.save_dir {
        // Saves inside the game folder would trip hot reload on every autosave.
        let d = std::path::absolute(d).map_err(|e| format!("--save-dir {}: {e}", d.display()))?;
        if d.starts_with(&o.root) {
            return Err(format!("--save-dir {} is inside the game folder; put saves somewhere else", d.display()));
        }
        o.save_dir = Some(d);
    }
    Ok(o)
}

/// Windows canonicalizes to `\\?\C:\...`, which Lua's file functions don't
/// understand; drop the prefix when it's a plain drive path.
fn plain_path(p: PathBuf) -> PathBuf {
    match p.to_str().and_then(|s| s.strip_prefix(r"\\?\")) {
        Some(rest) if !rest.starts_with("UNC") => PathBuf::from(rest),
        _ => p,
    }
}

/// Saves go to the user's data folder when a person is playing. Headless,
/// screenshot and timed runs keep them in memory so tests and docs never read
/// or overwrite real progress, unless `--save-dir` names a folder.
fn save_store(o: &Opts) -> save::Store {
    let dir = match &o.save_dir {
        Some(d) => Some(d.clone()),
        None if o.headless.is_some() || o.screenshot.is_some() || o.frames.is_some() => None,
        None => {
            let game = o
                .root
                .file_name()
                .map_or("game".into(), |n| n.to_string_lossy().replace(|c: char| !c.is_ascii_alphanumeric() && c != '-', "_"));
            let d = save::user_dir(&game);
            if d.is_none() {
                eprintln!("saves: no user data folder found; progress won't be kept");
            }
            d
        }
    };
    match dir {
        Some(d) => {
            println!("saves: {}", d.display());
            save::Store::Dir(d)
        }
        None => save::Store::default(),
    }
}

/// Newest modification time of any file under `dir`.
fn newest_mtime(dir: &Path) -> Option<SystemTime> {
    let mut best = None;
    for entry in std::fs::read_dir(dir).ok()?.flatten() {
        let t = if entry.file_type().ok()?.is_dir() { newest_mtime(&entry.path()) } else { entry.metadata().ok()?.modified().ok() };
        best = best.max(t);
    }
    best
}

/// The frame event's dt: the time since the last frame, zero while paused
/// so anything animated per frame holds still.
fn frame_dt(eng: &Shared, dt: f32) -> f32 {
    if eng.borrow().paused { 0.0 } else { dt }
}

/// Key events go to the game, or to menus while it's paused.
fn key_event(eng: &Shared) -> &'static str {
    if eng.borrow().paused { "menu_key" } else { "key" }
}

/// One simulation tick: scripts think, physics moves, contacts are reported.
fn tick(eng: &Shared, script: &Script) {
    let dt = (1.0 / TICK_HZ) as f32;
    eng.borrow_mut().snapshot();
    script.emit(eng, "tick", dt);
    let contacts = eng.borrow_mut().step(dt);
    for c in contacts {
        script.emit(eng, "contact", (c.a, c.b, c.speed));
    }
}

fn load(eng: &Shared) -> Option<Script> {
    match Script::load(eng) {
        Ok(s) => {
            s.emit(eng, "init", ());
            Some(s)
        }
        Err(e) => {
            script::report(eng, "load", &e.to_string());
            None
        }
    }
}

fn run_headless(root: PathBuf, ticks: u64, saves: save::Store, fullscreen: bool) {
    let mixer = Arc::new(Mutex::new(audio::Mixer::default()));
    let eng: Shared = Rc::new(RefCell::new(Engine::new(root, mixer.clone())));
    eng.borrow_mut().saves = saves;
    eng.borrow_mut().fullscreen = fullscreen;
    // Mix (and discard) audio in step with ticks so headless runs exercise it too.
    let mut sink = vec![0.0f32; 2 * (audio::RATE as f64 / TICK_HZ) as usize];
    let mut script = load(&eng);
    let t0 = std::time::Instant::now();
    let mut n = 0;
    while let Some(s) = &script {
        if n >= ticks {
            s.emit(&eng, "quit", ());
            break;
        }
        n += 1;
        if !eng.borrow().paused {
            tick(&eng, s);
        }
        eng.borrow_mut().hud.clear();
        s.emit(&eng, "frame", frame_dt(&eng, (1.0 / TICK_HZ) as f32));
        eng.borrow_mut().update_audio();
        mixer.lock().unwrap().mix(&mut sink);
        if eng.borrow().quit {
            s.emit(&eng, "quit", ());
            break;
        }
        // engine.reload(), as the windowed game does it: the old scripts
        // save, then everything starts afresh, saves kept.
        if std::mem::take(&mut eng.borrow_mut().reload) {
            s.emit(&eng, "quit", ());
            let (root, saves, fullscreen) = {
                let mut e = eng.borrow_mut();
                (e.root.clone(), std::mem::take(&mut e.saves), e.fullscreen)
            };
            drop(script.take()); // the old Lua state goes before the new one is built
            *eng.borrow_mut() = Engine::new(root, mixer.clone());
            eng.borrow_mut().saves = saves;
            eng.borrow_mut().fullscreen = fullscreen;
            script = load(&eng);
            println!("reloaded");
        }
    }
    let e = eng.borrow();
    let mesh_mb = e.meshes.iter().map(|m| m.verts.len() * size_of::<mesh::Vertex>() + m.indices.len() * 4).sum::<usize>() as f64 / 1e6;
    println!(
        "headless: {} ticks in {:.1} ms, {} entities, {} meshes ({mesh_mb:.0} MB)",
        e.tick,
        t0.elapsed().as_secs_f64() * 1e3,
        e.live,
        e.meshes.len()
    );
    if e.error.is_some() {
        std::process::exit(1);
    }
}

struct App {
    eng: Shared,
    _audio: Option<audio::Output>,
    script: Option<Script>,
    gfx: render::Renderer,
    acc: f64,
    last: f64,
    /// Seconds between the last two frames.
    frame_time: f32,
    watch_at: f64,
    mtime: Option<SystemTime>,
    reload: bool,
    frame: u64,
    grabbed: bool,
    shot: Option<(String, u64)>,
    quit_after: Option<u64>,
    ticks_per_frame: u32,
    closed: bool,
    fullscreen: bool,
    pads: pad::Pads,
    target_frame: Option<Duration>,
    frame_start: Instant,
    failed: Rc<Cell<bool>>,
}

impl App {
    /// Tell scripts the game is closing (once), so they can save.
    fn close(&mut self) {
        if !std::mem::replace(&mut self.closed, true)
            && let Some(s) = &self.script
        {
            s.emit(&self.eng, "quit", ());
        }
        self.failed.set(self.failed.get() || self.eng.borrow().error.is_some());
    }

    fn reload(&mut self) {
        // The old scripts get to save before they're replaced.
        if let Some(s) = &self.script {
            s.emit(&self.eng, "quit", ());
        }
        self.failed.set(self.failed.get() || self.eng.borrow().error.is_some());
        let (root, screen, mixer, saves, fullscreen) = {
            let mut e = self.eng.borrow_mut();
            (e.root.clone(), e.screen, e.mixer.clone(), std::mem::take(&mut e.saves), e.fullscreen)
        };
        // Drop the old Lua state before building the new one.
        self.script = None;
        *self.eng.borrow_mut() = Engine::new(root, mixer);
        self.eng.borrow_mut().screen = screen;
        self.eng.borrow_mut().saves = saves;
        self.eng.borrow_mut().fullscreen = fullscreen;
        self.script = load(&self.eng);
        self.acc = 0.0;
        println!("reloaded");
    }
}

impl EventHandler for App {
    fn update(&mut self) {
        let now = miniquad::date::now();
        let dt = (now - self.last).min(0.25);
        self.last = now;
        self.frame_time = dt as f32;
        if now >= self.watch_at {
            self.watch_at = now + 0.5;
            let m = newest_mtime(&self.eng.borrow().root);
            if m != self.mtime {
                self.reload |= self.mtime.is_some();
                self.mtime = m;
            }
        }
        let asked = std::mem::take(&mut self.eng.borrow_mut().reload);
        if std::mem::take(&mut self.reload) || asked {
            self.reload();
        }
        self.poll_pads(now);
        let Some(script) = &self.script else { return };
        if self.eng.borrow().paused {
            self.acc = 0.0;
            return;
        }
        // Fixed timestep; in screenshot mode advance exactly one tick per frame.
        self.acc += if self.shot.is_some() || self.quit_after.is_some() { self.ticks_per_frame as f64 / TICK_HZ } else { dt };
        while self.acc >= 1.0 / TICK_HZ {
            self.acc -= 1.0 / TICK_HZ;
            tick(&self.eng, script);
        }
    }

    fn draw(&mut self) {
        let (w, h) = window::screen_size();
        {
            let mut e = self.eng.borrow_mut();
            e.screen = (w, h);
            e.hud.clear();
        }
        // Draw between the last two ticks, as far along as the clock is
        // toward the next (screenshots step whole ticks, so are always at one).
        let fixed = self.shot.is_some() || self.quit_after.is_some();
        let along = if fixed || self.eng.borrow().paused { 1.0 } else { (self.acc * TICK_HZ).clamp(0.0, 1.0) as f32 };
        let saved = self.eng.borrow_mut().blend(along);
        if let Some(s) = &self.script {
            let dt = if fixed { self.ticks_per_frame as f32 / TICK_HZ as f32 } else { self.frame_time };
            s.emit(&self.eng, "frame", frame_dt(&self.eng, dt));
        }
        self.eng.borrow_mut().update_audio();
        {
            let mut e = self.eng.borrow_mut();
            e.mouse_delta = (0.0, 0.0);
            if let Some(err) = e.error.clone() {
                let msg: String = err.lines().take(12).collect::<Vec<_>>().join("\n");
                e.hud_rect(0.0, 0.0, w, 20.0 + 18.0 * msg.lines().count() as f32, [0.3, 0.0, 0.0, 0.85]);
                e.hud_text(&msg, 10.0, 10.0, 12.0, [1.0, 0.8, 0.8, 1.0]);
            }
            if e.fullscreen != self.fullscreen {
                self.fullscreen = e.fullscreen;
                window::set_fullscreen(self.fullscreen);
            }
            if e.mouse_grab != self.grabbed {
                self.grabbed = e.mouse_grab;
                window::set_cursor_grab(self.grabbed);
                window::show_mouse(!self.grabbed);
            }
        }
        self.gfx.draw(&self.eng.borrow());
        let uploaded = self.gfx.take_uploaded();
        self.eng.borrow_mut().release_uploaded(&uploaded);
        self.eng.borrow_mut().unblend(saved);
        self.frame += 1;
        if let Some((path, at)) = &self.shot
            && self.frame == *at
        {
            match self.gfx.screenshot(path, w as i32, h as i32) {
                Ok(()) => println!("screenshot: {path} ({} draw calls)", self.gfx.draw_calls),
                Err(e) => eprintln!("screenshot failed: {e}"),
            }
            self.close();
            window::order_quit();
        }
        self.gfx.commit();
        if let Some(target) = self.target_frame {
            let elapsed = self.frame_start.elapsed();
            if elapsed < target {
                std::thread::sleep(target - elapsed);
            }
            self.frame_start = Instant::now();
        }
        if self.eng.borrow().quit || self.quit_after == Some(self.frame) {
            self.close();
            window::order_quit();
        }
    }

    fn key_down_event(&mut self, key: KeyCode, _m: KeyMods, repeat: bool) {
        if key == KeyCode::F5 {
            self.reload = true;
            return;
        }
        if key == KeyCode::Escape {
            self.eng.borrow_mut().mouse_grab = false;
        }
        let name = key_name(key);
        if repeat || name.is_empty() {
            return;
        }
        self.press(name, true);
    }

    fn key_up_event(&mut self, key: KeyCode, _m: KeyMods) {
        self.press(key_name(key), false);
    }

    fn mouse_button_down_event(&mut self, b: MouseButton, _x: f32, _y: f32) {
        self.mouse_button(b, true);
    }

    fn mouse_button_up_event(&mut self, b: MouseButton, _x: f32, _y: f32) {
        self.mouse_button(b, false);
    }

    fn mouse_motion_event(&mut self, x: f32, y: f32) {
        self.eng.borrow_mut().cursor = (x, y);
    }

    fn window_minimized_event(&mut self) {
        if let Some(s) = &self.script {
            s.emit(&self.eng, "minimized", ());
        }
    }

    fn quit_requested_event(&mut self) {
        self.close();
    }

    fn raw_mouse_motion(&mut self, dx: f32, dy: f32) {
        let mut e = self.eng.borrow_mut();
        e.mouse_delta.0 += dx;
        e.mouse_delta.1 += dy;
    }
}

impl App {
    /// A key, mouse button or pad button went down or up.
    fn press(&mut self, name: &str, down: bool) {
        if down {
            self.eng.borrow_mut().keys.insert(name.to_string());
        } else {
            self.eng.borrow_mut().keys.remove(name);
        }
        if let Some(s) = &self.script {
            s.emit(&self.eng, key_event(&self.eng), (name, down));
        }
    }

    /// Gamepad buttons arrive as keys; sticks and triggers are read with engine.pad().
    fn poll_pads(&mut self, now: f64) {
        let mut events = Vec::new();
        self.pads.poll(now, &mut events);
        self.eng.borrow_mut().pad = self.pads.state().cloned();
        for ev in events {
            let (name, connected) = match ev {
                pad::Event::Button(name, down) => {
                    self.press(name, down);
                    continue;
                }
                pad::Event::Connected(name) => (name, true),
                pad::Event::Disconnected(name) => (name, false),
            };
            println!("gamepad {}: {name}", if connected { "connected" } else { "disconnected" });
            if let Some(s) = &self.script {
                s.emit(&self.eng, if connected { "pad_connected" } else { "pad_disconnected" }, name.as_str());
            }
        }
    }

    fn mouse_button(&mut self, b: MouseButton, down: bool) {
        let name = match b {
            MouseButton::Left => "mouse1",
            MouseButton::Right => "mouse2",
            MouseButton::Middle => "mouse3",
            MouseButton::Unknown => return,
        };
        self.press(name, down);
    }
}

fn main() {
    let o = match parse_args() {
        Ok(o) => o,
        Err(e) => {
            eprintln!("auto: {e}");
            std::process::exit(2);
        }
    };
    let saves = save_store(&o);
    if let Some(ticks) = o.headless {
        return run_headless(o.root, ticks, saves, o.fullscreen);
    }
    let conf = conf::Conf {
        window_title: "auto".into(),
        window_width: o.size.0,
        window_height: o.size.1,
        fullscreen: o.fullscreen,
        sample_count: 4,
        window_resizable: true,
        ..Default::default()
    };
    let (root, shot, frames, ticks_per_frame, fps, fullscreen) =
        (o.root, o.screenshot, o.frames, o.ticks_per_frame.max(1), o.fps, o.fullscreen);
    let target_frame = if shot.is_none() && frames.is_none() && fps > 0 { Some(Duration::from_secs_f64(1.0 / fps as f64)) } else { None };
    let failed = Rc::new(Cell::new(false));
    let result = failed.clone();
    miniquad::start(conf, move || {
        let mixer = Arc::new(Mutex::new(audio::Mixer::default()));
        let out = audio::start(mixer.clone());
        let eng: Shared = Rc::new(RefCell::new(Engine::new(root.clone(), mixer)));
        eng.borrow_mut().saves = saves;
        eng.borrow_mut().fullscreen = fullscreen;
        let script = load(&eng);
        let mtime = newest_mtime(&root);
        Box::new(App {
            eng,
            _audio: out,
            script,
            gfx: render::Renderer::new(),
            acc: 0.0,
            frame_time: (1.0 / TICK_HZ) as f32,
            last: miniquad::date::now(),
            watch_at: 0.0,
            mtime,
            reload: false,
            frame: 0,
            grabbed: false,
            quit_after: if shot.is_none() { frames } else { None },
            shot: shot.map(|p| (p, frames.unwrap_or(120))),
            ticks_per_frame,
            closed: false,
            fullscreen,
            pads: pad::Pads::new(),
            target_frame,
            frame_start: Instant::now(),
            failed,
        })
    });
    if result.get() {
        std::process::exit(1);
    }
}
```

```rs
impl Engine {
    /// Remember every pose before a tick.
    pub fn snapshot(&mut self) {
        for e in self.ents.iter_mut().flatten() {
            e.prev = Some((e.pos, e.rot));
        }
    }

    /// Move everything `t` (0..1) of the way from its pose before the last
    /// tick to its pose now, for drawing a frame that falls between ticks.
    /// Returns what `unblend` puts back.
    pub fn blend(&mut self, t: f32) -> Vec<(usize, [Vec3; 2], [Quat; 2])> {
        let mut saved = Vec::new();
        for (i, e) in self.ents.iter_mut().enumerate() {
            let Some(e) = e else { continue };
            let Some((p, r)) = e.prev else { continue };
            if p == e.pos && r == e.rot {
                continue;
            }
            let (bp, br) = (p + (e.pos - p) * t, r.nlerp(e.rot, t));
            saved.push((i, [e.pos, bp], [e.rot, br]));
            e.pos = bp;
            e.rot = br;
        }
        saved
    }

    /// Back to the ticked poses, except where the frame moved something itself.
    pub fn unblend(&mut self, saved: Vec<(usize, [Vec3; 2], [Quat; 2])>) {
        for (i, [p, bp], [r, br]) in saved {
            if let Some(Some(e)) = self.ents.get_mut(i) {
                if e.pos == bp {
                    e.pos = p;
                }
                if e.rot == br {
                    e.rot = r;
                }
            }
        }
    }

    pub fn new(root: PathBuf, mixer: Arc<Mutex<Mixer>>) -> Engine {
        mixer.lock().unwrap().stop_all();
        let mut e = Engine {
            root,
            ents: Vec::new(),

            warm: WarmCache::new(),
            broad: Broad::default(),
            touched: Vec::new(),
            tagged: HashMap::new(),
            terrain: None,
            terrain_chunks: Vec::new(),
            meshes: Vec::new(),
            mesh_ids: HashMap::new(),
            mesh_revs: Vec::new(),
            gpu_only: HashSet::new(),
            epoch: EPOCH.fetch_add(1, Ordering::Relaxed),
            camera: Camera { eye: v3(0., 10., -20.), target: Vec3::ZERO, fov: 70f32.to_radians() },
            env: Env {
                sky: [0.55, 0.7, 0.85],
                sky_top: [0.32, 0.52, 0.85],
                sun_color: [0.9, 0.84, 0.72],
                ambient: [0.42, 0.45, 0.52],
                ground: [0.12, 0.12, 0.12],
                glow: 0.0,
                shadows: true,
                fog_start: 80.0,
                fog_end: 260.0,
                sun: v3(0.4, 0.8, 0.3).norm(),
                gravity: -20.0,
            },
            lights: Vec::new(),
            keys: HashSet::new(),
            mouse_delta: (0.0, 0.0),
            mouse_grab: false,
            screen: (1280.0, 720.0),
            hud: Vec::new(),
            time: 0.0,
            tick: 0,
            quit: false,
            reload: false,
            error: None,
            mixer,
            sounds: HashMap::new(),
            follow: Vec::new(),
            saves: Default::default(),
            paused: false,
            fullscreen: false,

            pad: None,
        };
        for (name, m) in mesh::builtins() {
            e.add_mesh(name, m);
        }
        e
    }

    /// Register (or replace) a named mesh. Returns its id.
    pub fn add_mesh(&mut self, name: &str, m: Mesh) -> usize {
        if let Some(&id) = self.mesh_ids.get(name) {
            self.meshes[id] = m;
            self.mesh_revs[id] += 1;
            self.gpu_only.remove(&id);
            return id;
        }
        self.meshes.push(m);
        self.mesh_revs.push(0);
        self.mesh_ids.insert(name.to_string(), self.meshes.len() - 1);
        self.meshes.len() - 1
    }

    /// Drop the CPU copy of meshes the renderer has uploaded. Only big ones
    /// go (baked roads, blocks, terrain): they are most of a city's memory and
    /// nothing builds other meshes from them. Replacing one uploads it again.
    pub fn release_uploaded(&mut self, ids: &[usize]) {
        for &id in ids {
            if self.meshes[id].verts.len() >= GPU_ONLY_VERTS {
                self.meshes[id] = Mesh::default();
                self.gpu_only.insert(id);
            }
        }
    }

    pub fn spawn(&mut self, e: Entity) -> u32 {
        let id = self.ents.len() as u32;
        self.tagged.entry(e.tag.clone()).or_default().push(id);
        self.ents.push(Some(e));
        self.live += 1;
        id
    }

    pub fn despawn(&mut self, id: u32) {
        if let Some(slot) = self.ents.get_mut(id as usize)
            && let Some(e) = slot.take()
        {
            self.live -= 1;
            if let Some(ids) = self.tagged.get_mut(&e.tag)
                && let Ok(k) = ids.binary_search(&id)
            {
                ids.remove(k);
            }
        }
    }

    /// Live entities with this tag, in spawn order.
    pub fn find(&self, tag: &str) -> Vec<u32> {
        self.tagged.get(tag).cloned().unwrap_or_default()
    }

    pub fn get(&self, id: u32) -> Option<&Entity> {
        self.ents.get(id as usize)?.as_ref()
    }

    pub fn get_mut(&mut self, id: u32) -> Option<&mut Entity> {
        self.ents.get_mut(id as usize)?.as_mut()
    }

    /// For moving or turning an entity: notes a static, so rays still find it there.
    pub fn pose_mut(&mut self, id: u32) -> Option<&mut Entity> {
        let en = self.ents.get_mut(id as usize)?.as_mut()?;
        if en.body.as_ref().is_some_and(|b| b.kind == physics::Kind::Static) {
            self.touched.push(id);
        }
        Some(en)
    }

    /// The nearest body or terrain a ray hits; see physics::raycast_near.
    pub fn raycast(&self, o: Vec3, dir: Vec3, max: f32, ignore: Option<u32>) -> Option<physics::RayHit> {
        let hit = physics::raycast_near(&self.ents, &self.broad, &self.touched, o, dir, max, ignore);
        let Some(t) = self.terrain.as_ref().filter(|t| Some(t.id as u32) != ignore) else { return hit };
        let reach = hit.as_ref().map_or(max, |h| h.dist);
        match t.ray(o, dir, reach) {
            Some((dist, normal)) => Some(physics::RayHit { id: t.id as u32, dist, point: o + dir * dist, normal }),
            None => hit,
        }
    }

    /// Replace the ground heightfield (or remove it with None). It's drawn
    /// in chunks of CHUNK cells, each cell colored by the average of its
    /// corners' `colors` (one rgb per height, or grass), and collides
    /// through an entity tagged "terrain".
    pub fn set_terrain(&mut self, terrain: Option<Terrain>, colors: Option<Vec<[f32; 3]>>) {
        const CHUNK: usize = 64;
        if let Some(old) = self.terrain.take() {
            self.despawn(old.id as u32);
        }
        for id in std::mem::take(&mut self.terrain_chunks) {
            self.despawn(id);
        }
        let Some(mut t) = terrain else { return };
        let mut holder = Entity::new(v3(0.0, -10_000.0, 0.0));
        holder.tag = "terrain".to_string();
        holder.visible = false;
        holder.body = Some(Body::new(physics::Kind::Static, Vec3::ZERO));
        t.id = self.spawn(holder) as usize;
        let grass = [0.3, 0.42, 0.24];
        let col = |i: usize, j: usize| colors.as_ref().map_or(grass, |c| c[j * t.w + i]);
        let point = |i: usize, j: usize| v3(t.x0 + i as f32 * t.step, t.at(i, j), t.z0 + j as f32 * t.step);
        let avg = |cs: [[f32; 3]; 3]| {
            let c = |k: usize| (cs[0][k] + cs[1][k] + cs[2][k]) / 3.0;
            v3(c(0), c(1), c(2))
        };
        for cj in (0..t.d - 1).step_by(CHUNK) {
            for ci in (0..t.w - 1).step_by(CHUNK) {
                let mut m = Mesh::default();
                for j in cj..(cj + CHUNK).min(t.d - 1) {
                    for i in ci..(ci + CHUNK).min(t.w - 1) {
                        // Wound so the faces point up; the same split as Terrain::surface.
                        let (a, b, c, d) = ((i, j), (i, j + 1), (i + 1, j), (i + 1, j + 1));
                        m.tri(point(a.0, a.1), point(b.0, b.1), point(c.0, c.1), avg([col(a.0, a.1), col(b.0, b.1), col(c.0, c.1)]));
                        m.tri(point(c.0, c.1), point(b.0, b.1), point(d.0, d.1), avg([col(c.0, c.1), col(b.0, b.1), col(d.0, d.1)]));
                    }
                }
                let mesh = self.add_mesh(&format!("terrain_{ci}_{cj}"), m);
                let mut chunk = Entity::new(Vec3::ZERO);
                chunk.mesh = Some(mesh);
                chunk.tag = "terrain_mesh".to_string();
                let id = self.spawn(chunk);
                self.terrain_chunks.push(id);
            }
        }
        self.terrain = Some(t);
    }

    /// World-space position and direction of a light, if its anchor exists.
    pub fn light_world(&self, l: &Light) -> Option<(Vec3, Vec3)> {
        match l.follow {
            None => Some((l.pos, l.dir)),
            Some(id) => self.get(id).map(|e| (e.pos + e.rot.rotate(l.pos), e.rot.rotate(l.dir))),
        }
    }

    /// Pause or resume. Pausing returns the keys that were held, now released,
    /// so scripts can be told they went up (nothing stays "held" through a menu).
    pub fn set_paused(&mut self, on: bool) -> Vec<String> {
        let mut released = Vec::new();
        if on && !self.paused {
            released = self.keys.drain().collect();
            released.sort();
        }
        self.paused = on;
        self.mixer.lock().unwrap().paused = on;
        released
    }

    pub fn step(&mut self, dt: f32) -> Vec<Contact> {
        self.tick += 1;
        self.time += dt as f64;
        let contacts = physics::step(&mut self.ents, dt, self.env.gravity, &mut self.warm, &mut self.broad, self.terrain.as_ref());
        self.touched.clear();
        contacts
    }

    /// Put the listener at the camera and move entity-following voices.
    pub fn update_audio(&mut self) {
        let mut m = self.mixer.lock().unwrap();
        m.listener = self.camera.eye;
        m.listener_right = (self.camera.target - self.camera.eye).cross(Vec3::UP).norm();
        let ents = &self.ents;
        self.follow.retain(|&(voice, ent)| match ents.get(ent as usize).and_then(|e| e.as_ref()) {
            Some(e) if m.playing(voice) => {
                m.set(voice, None, None, Some(e.pos));
                true
            }
            _ => {
                m.stop(voice);
                false
            }
        });
    }

    pub fn hud_rect(&mut self, x: f32, y: f32, w: f32, h: f32, c: [f32; 4]) {
        for (px, py) in [(x, y), (x + w, y), (x + w, y + h), (x, y), (x + w, y + h), (x, y + h)] {
            self.hud.extend_from_slice(&[px, py, c[0], c[1], c[2], c[3]]);
        }
    }

    /// Bitmap text. `size` is the pixel height of a glyph cell (7 font pixels).
    pub fn hud_text(&mut self, s: &str, x: f32, y: f32, size: f32, c: [f32; 4]) {
        let px = size / 7.0;
        let (mut cx, mut cy) = (x, y);
        for ch in s.chars() {
            if ch == '\n' {
                cx = x;
                cy += size * 1.4;
                continue;
            }
            let mut pix = Vec::new();
            font::glyph(ch, |gx, gy| pix.push((gx, gy)));
            for (gx, gy) in pix {
                self.hud_rect(cx + gx as f32 * px, cy + gy as f32 * px, px, px, c);
            }
            cx += font::ADVANCE * px;
        }
    }

    pub fn text_width(s: &str, size: f32) -> f32 {
        let longest = s.lines().map(|l| l.chars().count()).max().unwrap_or(0);
        longest as f32 * font::ADVANCE * size / 7.0
    }
}
```

```rs
//! Rigid-body boxes. Oriented boxes collide via the separating axis test,
//! contacts are the penetrating corners, and a sequential-impulse solver
//! (accumulated, clamped normal + two-axis friction, Baumgarte bias) applies
//! both linear and angular response. Cars roll, crates tumble, ramps work.
//!
//! Bodies with `step > 0` walk up low ledges instead of bumping into them,
//! and `lock_rotation` keeps characters upright.

use crate::engine::Entity;
use crate::math::{Quat, Vec3, v3};
use std::collections::{HashMap, HashSet};

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Kind {
    Static,
    Dynamic,
    Kinematic,
}

#[derive(Clone, Debug)]
pub struct Body {
    pub kind: Kind,
    pub half: Vec3,
    pub vel: Vec3,
    pub ang_vel: Vec3,
    pub inv_mass: f32,
    /// Body-space diagonal inverse inertia.
    pub inv_inertia: Vec3,
    pub restitution: f32,
    pub friction: f32,
    pub gravity: f32,
    pub sensor: bool,
    pub grounded: bool,
    /// Ledges up to this height are stepped onto instead of blocking.
    pub step: f32,
    pub lock_rotation: bool,
}

impl Body {
    pub fn new(kind: Kind, half: Vec3) -> Body {
        let mut b = Body {
            kind,
            half,
            vel: Vec3::ZERO,
            ang_vel: Vec3::ZERO,
            inv_mass: 0.0,
            inv_inertia: Vec3::ZERO,
            restitution: 0.0,
            friction: 0.6,
            gravity: 1.0,
            sensor: false,
            grounded: false,
            step: 0.0,
            lock_rotation: false,
        };
        b.set_mass(1.0);
        b
    }

    /// Solid box inertia. Non-dynamic or rotation-locked bodies get none.
    pub fn set_mass(&mut self, m: f32) {
        if self.kind != Kind::Dynamic || m <= 0.0 {
            self.inv_mass = 0.0;
            self.inv_inertia = Vec3::ZERO;
            return;
        }
        self.inv_mass = 1.0 / m;
        let h = self.half;
        let i = |a: f32, b: f32| if self.lock_rotation { 0.0 } else { 3.0 / (m * (a * a + b * b)) };
        self.inv_inertia = v3(i(h.y, h.z), i(h.x, h.z), i(h.x, h.y));
    }

    fn inv_inertia_world(&self, q: Quat, v: Vec3) -> Vec3 {
        if self.inv_inertia == Vec3::ZERO {
            return Vec3::ZERO; // static, kinematic or rotation-locked
        }
        q.rotate(q.conj().rotate(v).mul(self.inv_inertia))
    }
}

/// World-space half extents of an oriented box's bounding box.
pub fn extents(half: Vec3, rot: Quat) -> Vec3 {
    let [a, b, c] = rot.axes();
    let (a, b, c) = (a.abs() * half.x, b.abs() * half.y, c.abs() * half.z);
    a + b + c
}

/// A heightfield: `w` x `d` heights, `step` meters apart, row by row from
/// (x0, z0) toward +x then +z. Each cell is two triangles split along its
/// (x+1, z)-(x, z+1) diagonal, and drawing and collision use that same
/// surface. `id` is its entity, the "b" side of terrain contacts.
pub struct Terrain {
    pub x0: f32,
    pub z0: f32,
    pub step: f32,
    pub w: usize,
    pub d: usize,
    pub heights: Vec<f32>,
    pub id: usize,
    pub top: f32,
    /// How far raised pavements and medians stand above the road beside them.
    pub kerb: f32,
    roads: Vec<RoadSeg>,
    /// Road segments by ROAD_CELL square, as ranges into `road_items`.
    road_cells: Vec<u32>,
    road_items: Vec<u32>,
    road_w: usize,
}

/// A stretch of road laid over the terrain: its centerline from `a` to `b`
/// (heights in y), flat across out to `half` meters each side. The asphalt
/// is `carriage` meters each side; past it, a side that's `raised` (a
/// pavement, a median) stands the terrain's `kerb` higher, out to `half`.
/// Side 0 is the one toward (dz, -dx), (dx, dz) being b - a.
#[derive(Clone, Copy)]
pub struct RoadSeg {
    pub a: Vec3,
    pub b: Vec3,
    pub half: f32,
    pub carriage: f32,
    pub raised: [bool; 2],
    /// Which road it's part of: a road follows its nearest piece, and
    /// different roads that overlap blend.
    pub road: u32,
}

const ROAD_CELL: f32 = 16.0;
/// Past a road's edge the surface eases back to the heightfield over this many meters.
const ROAD_BLEND: f32 = 2.0;

impl Terrain {
    pub fn new(x0: f32, z0: f32, step: f32, w: usize, d: usize, heights: Vec<f32>) -> Terrain {
        let top = heights.iter().copied().fold(f32::MIN, f32::max);
        Terrain {
            x0,
            z0,
            step,
            w,
            d,
            heights,
            id: usize::MAX,
            top,
            kerb: 0.0,
            roads: Vec::new(),
            road_cells: Vec::new(),
            road_items: Vec::new(),
            road_w: 0,
        }
    }

    /// Lay roads over the heightfield: on them the surface is the road's
    /// own, smooth along it and level across, instead of the grid's
    /// triangles; it eases back to the grid just past their edges.
    pub fn with_roads(mut self, roads: Vec<RoadSeg>) -> Terrain {
        let (rw, rd) =
            (((self.w - 1) as f32 * self.step / ROAD_CELL) as usize + 1, ((self.d - 1) as f32 * self.step / ROAD_CELL) as usize + 1);
        let mut lists: Vec<Vec<u32>> = vec![Vec::new(); rw * rd];
        let cell = |v: f32, n: usize| ((v / ROAD_CELL).floor().max(0.0) as usize).min(n - 1);
        for (k, r) in roads.iter().enumerate() {
            let pad = r.half + ROAD_BLEND;
            let (lx, hx) = (r.a.x.min(r.b.x) - pad - self.x0, r.a.x.max(r.b.x) + pad - self.x0);
            let (lz, hz) = (r.a.z.min(r.b.z) - pad - self.z0, r.a.z.max(r.b.z) + pad - self.z0);
            for j in cell(lz, rd)..=cell(hz, rd) {
                for i in cell(lx, rw)..=cell(hx, rw) {
                    lists[j * rw + i].push(k as u32);
                }
            }
            self.top = self.top.max(r.a.y).max(r.b.y);
        }
        self.road_cells = Vec::with_capacity(lists.len() + 1);
        self.road_items.clear();
        for l in &lists {
            self.road_cells.push(self.road_items.len() as u32);
            self.road_items.extend_from_slice(l);
        }
        self.road_cells.push(self.road_items.len() as u32);
        self.roads = roads;
        self.road_w = rw;
        // Where a road runs below the grid (in a sag, or down a trench into
        // a tunnel) the grid under it comes down to meet it, so the ground
        // drawn there doesn't cover the road.
        for j in 0..self.d {
            for i in 0..self.w {
                let (x, z) = (self.x0 + i as f32 * self.step, self.z0 + j as f32 * self.step);
                if let Some((on, rh, _)) = self.road(x, z)
                    && on >= 1.0
                    && rh - 0.15 < self.heights[j * self.w + i]
                {
                    self.heights[j * self.w + i] = rh - 0.15;
                }
            }
        }
        self
    }

    pub fn at(&self, i: usize, j: usize) -> f32 {
        self.heights[j * self.w + i]
    }

    /// How much of (x, z) is road (1 on one, easing to 0 over ROAD_BLEND
    /// past its edge), and the roads' height and normal there. Where roads
    /// overlap, as they do around junctions, they're averaged, nearer
    /// centerlines counting more, so one never steps onto another. On a
    /// raised side it's a kerb higher, unless it's on some road's asphalt
    /// (where one road's pavement meets another road at a junction).
    fn road(&self, x: f32, z: f32) -> Option<(f32, f32, Vec3)> {
        if self.roads.is_empty() {
            return None;
        }
        let (u, v) = ((x - self.x0) / ROAD_CELL, (z - self.z0) / ROAD_CELL);
        if !(u >= 0.0 && v >= 0.0) || u as usize >= self.road_w || (v as usize + 1) * self.road_w >= self.road_cells.len() {
            return None;
        }
        let c = v as usize * self.road_w + u as usize;
        // Per road: its pieces blended, the ones (x, z) is beside by far the
        // most (so inside a bend there's no crease where one piece hands over
        // to the next):
        // (road, weight, weighted height, weighted normal, nearest distance
        // squared, coverage, reach).
        let mut near = [(u32::MAX, 0.0f32, 0.0f32, Vec3::ZERO, f32::MAX, 0.0f32, 0.0f32); 8];
        let mut n = 0;
        let (mut asphalt, mut raised) = (false, false);
        for &k in &self.road_items[self.road_cells[c] as usize..self.road_cells[c + 1] as usize] {
            let r = &self.roads[k as usize];
            let (dx, dz) = (r.b.x - r.a.x, r.b.z - r.a.z);
            let l2 = dx * dx + dz * dz;
            let along = if l2 > 1e-9 { ((x - r.a.x) * dx + (z - r.a.z) * dz) / l2 } else { 0.0 };
            let t = along.clamp(0.0, 1.0);
            let d2 = (x - r.a.x - dx * t).powi(2) + (z - r.a.z - dz * t).powi(2);
            let on = 1.0 - (d2.sqrt() - r.half) / ROAD_BLEND;
            if on <= 0.0 {
                continue;
            }
            let g = if l2 > 1e-9 { (r.b.y - r.a.y) / l2 } else { 0.0 };
            let (h, nrm) = (r.a.y + (r.b.y - r.a.y) * t, v3(-dx * g, 1.0, -dz * g).norm());
            // Beside its own piece a point counts that piece almost alone;
            // a meter past a piece's end, it barely counts, nor toward the
            // edge of a piece across a bend.
            let past_end = (along - t).abs() * l2.sqrt();
            if d2 <= r.carriage * r.carriage {
                asphalt = true;
            } else if past_end < 1e-3 && d2 <= r.half * r.half {
                let side = ((x - r.a.x) * dz - (z - r.a.z) * dx < 0.0) as usize;
                raised |= r.raised[side];
            }
            let w = (1.0 - d2.sqrt() / (r.half + ROAD_BLEND)).powi(2) / (past_end * past_end + 0.01).powi(2);
            let i = match near[..n].iter().position(|p| p.0 == r.road) {
                Some(i) => i,
                None if n < near.len() => {
                    n += 1;
                    near[n - 1].0 = r.road;
                    n - 1
                }
                None => continue,
            };
            let p = &mut near[i];
            p.1 += w;
            p.2 += w * h;
            p.3 += nrm * w;
            p.4 = p.4.min(d2);
            p.5 = p.5.max(on.min(1.0));
            p.6 = r.half + ROAD_BLEND;
        }
        // Then the roads together, nearer centerlines counting more and
        // fading out smoothly toward each one's outer edge, so between two
        // the surface leans from one to the other without a step.
        let (mut cover, mut sw, mut sh, mut sn) = (0.0f32, 0.0f32, 0.0f32, Vec3::ZERO);
        for &(_, pw, ph, pn, d2, on, reach) in &near[..n] {
            let w = (1.0 - d2.sqrt() / reach).powi(2);
            cover = cover.max(on);
            sw += w;
            sh += w * ph / pw;
            sn += pn.norm() * w;
        }
        let kerb = if raised && !asphalt { self.kerb } else { 0.0 };
        (sw > 0.0).then(|| (cover, sh / sw + kerb, sn.norm()))
    }

    /// Height and upward normal at (x, z), roads included; None off the grid.
    pub fn surface(&self, x: f32, z: f32) -> Option<(f32, Vec3)> {
        let (h, n) = self.grid(x, z)?;
        Some(match self.road(x, z) {
            Some((on, rh, rn)) => (h + (rh - h) * on, (n * (1.0 - on) + rn * on).norm()),
            None => (h, n),
        })
    }

    /// The heightfield's own triangles at (x, z), as drawn.
    pub fn grid(&self, x: f32, z: f32) -> Option<(f32, Vec3)> {
        let (u, v) = ((x - self.x0) / self.step, (z - self.z0) / self.step);
        if !(u >= 0.0 && v >= 0.0) {
            return None;
        }
        let (i, j) = (u as usize, v as usize);
        if i + 1 >= self.w || j + 1 >= self.d {
            return None;
        }
        let (fx, fz) = (u - i as f32, v - j as f32);
        let (h00, h10, h01, h11) = (self.at(i, j), self.at(i + 1, j), self.at(i, j + 1), self.at(i + 1, j + 1));
        let (h, dx, dz) = if fx + fz <= 1.0 {
            (h00 + (h10 - h00) * fx + (h01 - h00) * fz, h10 - h00, h01 - h00)
        } else {
            (h11 + (h01 - h11) * (1.0 - fx) + (h10 - h11) * (1.0 - fz), h11 - h01, h11 - h10)
        };
        Some((h, v3(-dx / self.step, 1.0, -dz / self.step).norm()))
    }

    /// Where a ray coming from above first meets the surface: distance and
    /// normal. Marches half a cell at a time, then bisects.
    pub fn ray(&self, o: Vec3, dir: Vec3, max: f32) -> Option<(f32, Vec3)> {
        let below = |t: f32| {
            let p = o + dir * t;
            self.surface(p.x, p.z).map(|(h, _)| p.y < h)
        };
        let dt = self.step * 0.5;
        let (mut t0, mut t) = (0.0f32, 0.0f32);
        if below(0.0) == Some(true) {
            return None; // starts underground
        }
        while t < max {
            t = (t + dt).min(max);
            let p = o + dir * t;
            if p.y > self.top && dir.y >= 0.0 {
                return None; // above every hill and climbing
            }
            if below(t) == Some(true) {
                let (mut lo, mut hi) = (t0, t);
                for _ in 0..16 {
                    let mid = (lo + hi) * 0.5;
                    if below(mid) == Some(true) {
                        hi = mid;
                    } else {
                        lo = mid;
                    }
                }
                let p = o + dir * hi;
                return self.surface(p.x, p.z).map(|(_, n)| (hi, n));
            }
            t0 = t;
        }
        None
    }
}

pub struct Contact {
    pub a: u32,
    pub b: u32,
    /// Closing speed along the contact normal when they met (m/s).
    pub speed: f32,
}

/// Accumulated impulses from last step, keyed by (body a, body b, feature).
/// Starting the solver from them ("warm starting") is what lets stacks rest.
pub type WarmCache = HashMap<(usize, usize, u8), (f32, f32, f32)>;

struct Point {
    feature: u8,
    ra: Vec3,
    rb: Vec3,
    depth: f32,
    t1: Vec3,
    t2: Vec3,
    mass_n: f32,
    mass_t1: f32,
    mass_t2: f32,
    bounce: f32,
    pn: f32,
    pt1: f32,
    pt2: f32,
}

struct Manifold {
    a: usize,
    b: usize,
    n: Vec3, // from b to a
    mu: f32,
    points: Vec<Point>,
}

const ITERATIONS: usize = 10;
const SLOP: f32 = 0.01;
const BAUMGARTE: f32 = 0.2;

fn body(ents: &[Option<Entity>], i: usize) -> (&Entity, &Body) {
    let e = ents[i].as_ref().unwrap();
    (e, e.body.as_ref().unwrap())
}

/// Broad phase. Static bodies sit in a uniform grid over XZ that persists
/// between steps and is rebuilt only when one is added, removed or moved.
/// Dynamic and kinematic bodies, and anything huge (the ground), are checked
/// against everyone. Candidates come back in ascending entity order, the
/// order a full scan visits them, so the solver sees the same sequence.
#[derive(Default)]
pub struct Broad {
    cells: HashMap<(i32, i32), Vec<usize>>,
    big: Vec<usize>,
    /// Each static's AABB, and the pose it had when the grid was built.
    boxes: Vec<Option<(Vec3, Vec3)>>,
    poses: Vec<Option<[f32; 10]>>,
    statics: usize,
    pub rebuilds: u64,
    /// As of the last step: the moving bodies, and how many slots there were.
    movers: Vec<usize>,
    len: usize,
}

impl Broad {
    const CELL: f32 = 16.0;
    /// Bodies spanning more than this many cells skip the grid.
    const BIG: i32 = 6;

    fn range(c: Vec3, h: Vec3) -> (i32, i32, i32, i32) {
        let f = |v: f32| (v / Self::CELL).floor() as i32;
        (f(c.x - h.x), f(c.x + h.x), f(c.z - h.z), f(c.z + h.z))
    }

    fn pose(p: Vec3, r: Quat, h: Vec3) -> [f32; 10] {
        [p.x, p.y, p.z, r.x, r.y, r.z, r.w, h.x, h.y, h.z]
    }

    fn rebuild(&mut self, ents: &[Option<Entity>]) {
        self.cells.clear();
        self.big.clear();
        self.boxes = vec![None; ents.len()];
        self.poses = vec![None; ents.len()];
        self.statics = 0;
        self.rebuilds += 1;
        for (i, e) in ents.iter().enumerate() {
            let Some(e) = e else { continue };
            let Some(b) = &e.body else { continue };
            if b.kind != Kind::Static {
                continue;
            }
            let (c, h) = (e.pos, extents(b.half, e.rot));
            self.boxes[i] = Some((c, h));
            self.poses[i] = Some(Self::pose(e.pos, e.rot, b.half));
            self.statics += 1;
            let (x0, x1, z0, z1) = Self::range(c, h);
            if x1 - x0 > Self::BIG || z1 - z0 > Self::BIG {
                self.big.push(i);
                continue;
            }
            for x in x0..=x1 {
                for z in z0..=z1 {
                    self.cells.entry((x, z)).or_default().push(i);
                }
            }
        }
    }

    /// Entities near the box: statics from its cells, big ones and `movers`, sorted.
    fn near(&self, (c, h): (Vec3, Vec3), movers: &[usize], out: &mut Vec<usize>) {
        out.clear();
        out.extend_from_slice(&self.big);
        out.extend_from_slice(movers);
        let (x0, x1, z0, z1) = Self::range(c, h);
        for x in x0..=x1 {
            for z in z0..=z1 {
                if let Some(v) = self.cells.get(&(x, z)) {
                    out.extend_from_slice(v);
                }
            }
        }
        out.sort_unstable();
        out.dedup();
    }
}

pub fn step(
    ents: &mut [Option<Entity>],
    dt: f32,
    gravity: f32,
    warm: &mut WarmCache,
    broad: &mut Broad,
    terrain: Option<&Terrain>,
) -> Vec<Contact> {
    // 1. Forces. Also notice whether any static body changed since the grid was built.
    let mut dynamic = Vec::new();
    let mut movers = Vec::new();
    let (mut statics, mut stale) = (0, false);
    for (i, e) in ents.iter_mut().enumerate() {
        let Some(e) = e else { continue };
        let Some(b) = &mut e.body else { continue };
        if b.kind == Kind::Static {
            statics += 1;
            stale = stale || broad.poses.get(i).copied().flatten() != Some(Broad::pose(e.pos, e.rot, b.half));
            continue;
        }
        movers.push(i);
        if b.kind == Kind::Dynamic {
            b.vel.y += gravity * b.gravity * dt;
            b.ang_vel = b.ang_vel * (1.0 / (1.0 + 0.3 * dt));
            if b.lock_rotation {
                b.ang_vel = Vec3::ZERO;
            }
            b.grounded = false;
            dynamic.push(i);
        }
    }
    if stale || statics != broad.statics {
        broad.rebuild(ents);
    }
    broad.movers.clone_from(&movers);
    broad.len = ents.len();

    // 2. Narrow phase. Moving bodies' AABBs are computed fresh; statics' come from the grid.
    let aabb = |e: &Entity| (e.pos, extents(e.body.as_ref().unwrap().half, e.rot));
    let mut moving: HashMap<usize, (Vec3, Vec3)> = HashMap::with_capacity(movers.len());
    for &i in &movers {
        moving.insert(i, aabb(ents[i].as_ref().unwrap()));
    }
    let bounds = |j: usize| moving.get(&j).copied().or_else(|| broad.boxes[j]).unwrap();
    let mut manifolds = Vec::new();
    let mut touching = Vec::new();
    let mut seen = HashSet::new();
    let mut near = Vec::new();
    for &i in &dynamic {
        let bi = moving[&i];
        broad.near(bi, &movers, &mut near);
        for &j in &near {
            if i == j {
                continue;
            }
            let ((ca, ha), (cb, hb)) = (bi, bounds(j));
            let (d, h) = ((ca - cb).abs(), ha + hb);
            if d.x >= h.x || d.y >= h.y || d.z >= h.z {
                continue;
            }
            let b_dynamic = body(ents, j).1.kind == Kind::Dynamic;
            if b_dynamic && j < i || !seen.insert((i.min(j), i.max(j))) {
                continue;
            }
            let (ea, ba) = body(ents, i);
            let (eb, bb) = body(ents, j);
            let Some((n, pts)) = collide(ea.pos, ea.rot, ba.half, eb.pos, eb.rot, bb.half) else { continue };
            if ba.sensor || bb.sensor {
                touching.push((i, j, 0.0));
                continue;
            }
            // Step-up: a low ledge met side-on lifts the body instead of walling it.
            // Narrow vertical posts and poles (e.g. lamp posts, traffic signs) must not act
            // as step ramps for vehicles; require the obstacle to have sufficient horizontal footprint.
            let b_is_pole = hb.x < 0.25 && hb.z < 0.25 && hb.y > 1.5;
            if !b_dynamic && !b_is_pole && ba.step > 0.0 && n.y.abs() < 0.5 {
                let rise = (cb.y + hb.y) - (ca.y - ha.y);
                if rise > 0.0 && rise <= ba.step && ca.y > cb.y + hb.y {
                    let e = ents[i].as_mut().unwrap();
                    e.pos.y += rise + 0.01;
                    let b = e.body.as_mut().unwrap();
                    b.vel.y = b.vel.y.max(0.0);
                    b.grounded = true;
                    touching.push((i, j, 0.0));
                    continue;
                }
            }
            let mu = (ba.friction * bb.friction).sqrt();
            let e_rest = ba.restitution.max(bb.restitution);
            let mut m = Manifold { a: i, b: j, n, mu, points: Vec::new() };
            let (t1, t2) = tangents(n);
            let mut closing: f32 = 0.0;
            for (feature, p, depth) in pts {
                let (ra, rb) = (p - ea.pos, p - eb.pos);
                let k = |dir: Vec3| {
                    let ia = ba.inv_mass + ba.inv_inertia_world(ea.rot, ra.cross(dir)).cross(ra).dot(dir);
                    let ib = bb.inv_mass + bb.inv_inertia_world(eb.rot, rb.cross(dir)).cross(rb).dot(dir);
                    1.0 / (ia + ib).max(1e-9)
                };
                let vn = rel_vel(ba, bb, ra, rb).dot(n);
                closing = closing.max(-vn);
                let bounce = if vn < -1.0 { -e_rest * vn } else { 0.0 };
                let (pn, pt1, pt2) = warm.get(&(i, j, feature)).copied().unwrap_or_default();
                m.points.push(Point { feature, ra, rb, depth, t1, t2, mass_n: k(n), mass_t1: k(t1), mass_t2: k(t2), bounce, pn, pt1, pt2 });
            }
            touching.push((i, j, closing));
            if n.y > 0.7 {
                ents[i].as_mut().unwrap().body.as_mut().unwrap().grounded = true;
            } else if n.y < -0.7 && b_dynamic {
                ents[j].as_mut().unwrap().body.as_mut().unwrap().grounded = true;
            }
            manifolds.push(m);
        }
    }

    // 2b. Terrain: each corner of a moving box that's below the surface
    // pushes off it along the surface's normal there, one contact each.
    if let Some(t) = terrain
        && ents.get(t.id).is_some_and(|e| e.is_some())
    {
        let mut grounded = Vec::new();
        for &i in &dynamic {
            let (ea, ba) = body(ents, i);
            let (eb, bb) = body(ents, t.id);
            if ba.sensor || ea.pos.y - ba.half.len() > t.top {
                continue;
            }
            let mut closing: f32 = 0.0;
            let mut hit = false;
            for (k, p) in corners(ea.pos, &ea.rot.axes(), ba.half).into_iter().enumerate() {
                let Some((h, n)) = t.surface(p.x, p.z) else { continue };
                if p.y >= h {
                    continue;
                }
                let feature = 24 + k as u8;
                let (ra, rb) = (p - ea.pos, p - eb.pos);
                let kk = |dir: Vec3| {
                    let ia = ba.inv_mass + ba.inv_inertia_world(ea.rot, ra.cross(dir)).cross(ra).dot(dir);
                    1.0 / ia.max(1e-9)
                };
                let vn = rel_vel(ba, bb, ra, rb).dot(n);
                closing = closing.max(-vn);
                let bounce = if vn < -1.0 { -ba.restitution * vn } else { 0.0 };
                let (pn, pt1, pt2) = warm.get(&(i, t.id, feature)).copied().unwrap_or_default();
                let (t1, t2) = tangents(n);
                let depth = (h - p.y) * n.y;
                let point = Point { feature, ra, rb, depth, t1, t2, mass_n: kk(n), mass_t1: kk(t1), mass_t2: kk(t2), bounce, pn, pt1, pt2 };
                manifolds.push(Manifold { a: i, b: t.id, n, mu: (ba.friction * bb.friction).sqrt(), points: vec![point] });
                hit = true;
                if n.y > 0.7 {
                    grounded.push(i);
                }
            }
            if hit {
                touching.push((i, t.id, closing));
            }
        }
        for i in grounded {
            ents[i].as_mut().unwrap().body.as_mut().unwrap().grounded = true;
        }
    }

    // 3. Sequential impulses, starting from last step's.
    for m in &manifolds {
        warm_start(ents, m);
    }
    for _ in 0..ITERATIONS {
        for m in &mut manifolds {
            solve(ents, m, dt);
        }
    }

    warm.clear();
    for m in &manifolds {
        for p in &m.points {
            warm.insert((m.a, m.b, p.feature), (p.pn, p.pt1, p.pt2));
        }
    }

    // 4. Integrate.
    for e in ents.iter_mut().flatten() {
        let Some(b) = &e.body else { continue };
        match b.kind {
            Kind::Static => {}
            _ => {
                e.pos += b.vel * dt;
                if b.ang_vel != Vec3::ZERO {
                    e.rot = e.rot.integrate(b.ang_vel, dt);
                }
            }
        }
    }

    touching.into_iter().map(|(a, b, speed)| Contact { a: a.min(b) as u32, b: a.max(b) as u32, speed }).collect()
}

fn rel_vel(a: &Body, b: &Body, ra: Vec3, rb: Vec3) -> Vec3 {
    (a.vel + a.ang_vel.cross(ra)) - (b.vel + b.ang_vel.cross(rb))
}

fn tangents(n: Vec3) -> (Vec3, Vec3) {
    let t1 = if n.x.abs() > 0.57 { v3(n.y, -n.x, 0.0) } else { v3(0.0, n.z, -n.y) }.norm();
    (t1, n.cross(t1))
}

/// Both bodies of a manifold, and a function applying an impulse at a contact.
fn bodies<'a>(
    ents: &'a mut [Option<Entity>],
    m: &Manifold,
) -> (&'a mut Body, &'a mut Body, impl Fn(&mut Body, &mut Body, Vec3, Vec3, Vec3) + use<>) {
    let (ea, eb) = pair_mut(ents, m.a, m.b);
    let (ea, eb) = (ea.as_mut().unwrap(), eb.as_mut().unwrap());
    let (qa, qb) = (ea.rot, eb.rot);
    let (a, b) = (ea.body.as_mut().unwrap(), eb.body.as_mut().unwrap());
    let b_moves = b.kind == Kind::Dynamic;
    let apply = move |a: &mut Body, b: &mut Body, ra: Vec3, rb: Vec3, imp: Vec3| {
        a.vel += imp * a.inv_mass;
        a.ang_vel += a.inv_inertia_world(qa, ra.cross(imp));
        if b_moves {
            b.vel -= imp * b.inv_mass;
            b.ang_vel -= b.inv_inertia_world(qb, rb.cross(imp));
        }
    };
    (a, b, apply)
}

fn warm_start(ents: &mut [Option<Entity>], m: &Manifold) {
    let (a, b, apply) = bodies(ents, m);
    for p in &m.points {
        apply(a, b, p.ra, p.rb, m.n * p.pn + p.t1 * p.pt1 + p.t2 * p.pt2);
    }
}

fn solve(ents: &mut [Option<Entity>], m: &mut Manifold, dt: f32) {
    let (a, b, apply) = bodies(ents, m);
    for p in &mut m.points {
        // Normal: push apart, with a bias that removes penetration.
        let vn = rel_vel(a, b, p.ra, p.rb).dot(m.n);
        let bias = (BAUMGARTE / dt * (p.depth - SLOP).max(0.0)).max(p.bounce);
        let old = p.pn;
        p.pn = (old + (bias - vn) * p.mass_n).max(0.0);
        apply(a, b, p.ra, p.rb, m.n * (p.pn - old));

        // Friction, bounded by the normal impulse (box approximation of the cone).
        let lim = m.mu * p.pn;
        for (t, acc, mass) in [(p.t1, &mut p.pt1, p.mass_t1), (p.t2, &mut p.pt2, p.mass_t2)] {
            let vt = rel_vel(a, b, p.ra, p.rb).dot(t);
            let old = *acc;
            *acc = (old - vt * mass).clamp(-lim, lim);
            apply(a, b, p.ra, p.rb, t * (*acc - old));
        }
    }
}

/// (feature id, world point, penetration depth). Feature ids are stable
/// across steps (which corner of which box) so impulses can be warm-started.
pub type ContactPoint = (u8, Vec3, f32);

/// Separating axis test for two oriented boxes. Returns the contact normal
/// (pointing from b to a) and contact points with their depths.
pub fn collide(ca: Vec3, qa: Quat, ha: Vec3, cb: Vec3, qb: Quat, hb: Vec3) -> Option<(Vec3, Vec<ContactPoint>)> {
    let (aa, ab) = (qa.axes(), qb.axes());
    let d = ca - cb;
    let radius = |ax: &[Vec3; 3], h: Vec3, l: Vec3| ax[0].dot(l).abs() * h.x + ax[1].dot(l).abs() * h.y + ax[2].dot(l).abs() * h.z;
    let mut best: Option<(f32, Vec3)> = None;
    let mut test = |l: Vec3, is_edge: bool| -> bool {
        let len = l.len();
        if len < 1e-4 {
            return true; // parallel edges: covered by face axes
        }
        let l = l * (1.0 / len);
        let overlap = radius(&aa, ha, l) + radius(&ab, hb, l) - d.dot(l).abs();
        if overlap < 0.0 {
            return false;
        }
        // Prefer face axes; edge axes must be clearly better to win.
        let score = if is_edge { overlap * 1.05 + 0.01 } else { overlap };
        if best.is_none_or(|(s, _)| score < s) {
            best = Some((score, if d.dot(l) < 0.0 { -l } else { l }));
        }
        true
    };
    for l in aa.iter().chain(ab.iter()) {
        if !test(*l, false) {
            return None;
        }
    }
    for x in &aa {
        for y in &ab {
            if !test(x.cross(*y), true) {
                return None;
            }
        }
    }
    let (_, n) = best?;

    // Contacts: corners of each box that are inside the other.
    let mut pts = Vec::new();
    let top_b = cb.dot(n) + radius(&ab, hb, n);
    let bottom_a = ca.dot(n) - radius(&aa, ha, n);
    for (k, p) in corners(ca, &aa, ha).into_iter().enumerate() {
        if inside(p, cb, &ab, hb) {
            pts.push((k as u8, p, top_b - p.dot(n)));
        }
    }
    for (k, p) in corners(cb, &ab, hb).into_iter().enumerate() {
        if inside(p, ca, &aa, ha) {
            pts.push((8 + k as u8, p, p.dot(n) - bottom_a));
        }
    }
    if pts.is_empty() {
        // Edge on edge: one contact between the two deepest features.
        let pa = support(ca, &aa, ha, -n);
        let pb = support(cb, &ab, hb, n);
        pts.push((16, (pa + pb) * 0.5, (pb - pa).dot(n).max(0.0)));
    }
    if pts.len() > 8 {
        pts.sort_by(|x, y| y.2.total_cmp(&x.2));
        pts.truncate(8);
    }
    Some((n, pts))
}

fn corners(c: Vec3, ax: &[Vec3; 3], h: Vec3) -> [Vec3; 8] {
    let mut out = [Vec3::ZERO; 8];
    for (i, o) in out.iter_mut().enumerate() {
        let s = |bit: usize| if i >> bit & 1 == 1 { 1.0 } else { -1.0 };
        *o = c + ax[0] * (h.x * s(0)) + ax[1] * (h.y * s(1)) + ax[2] * (h.z * s(2));
    }
    out
}

fn inside(p: Vec3, c: Vec3, ax: &[Vec3; 3], h: Vec3) -> bool {
    let d = p - c;
    let tol = 1e-3;
    d.dot(ax[0]).abs() <= h.x + tol && d.dot(ax[1]).abs() <= h.y + tol && d.dot(ax[2]).abs() <= h.z + tol
}

/// Corner of the box farthest along `dir`.
fn support(c: Vec3, ax: &[Vec3; 3], h: Vec3, dir: Vec3) -> Vec3 {
    let s = |a: Vec3, e: f32| a * (e * a.dot(dir).signum());
    c + s(ax[0], h.x) + s(ax[1], h.y) + s(ax[2], h.z)
}

fn pair_mut<T>(s: &mut [T], i: usize, j: usize) -> (&mut T, &mut T) {
    assert!(i != j);
    if i < j {
        let (l, r) = s.split_at_mut(j);
        (&mut l[i], &mut r[0])
    } else {
        let (l, r) = s.split_at_mut(i);
        (&mut r[0], &mut l[j])
    }
}

pub struct RayHit {
    pub id: u32,
    pub dist: f32,
    pub point: Vec3,
    pub normal: Vec3,
}

/// Slab test in each box's local frame. `dir` must be normalized.
pub fn raycast(ents: &[Option<Entity>], o: Vec3, dir: Vec3, max: f32, ignore: Option<u32>) -> Option<RayHit> {
    let mut best = None;
    for i in 0..ents.len() {
        ray_test(ents, i, o, dir, max, ignore, &mut best);
    }
    best
}

/// The same ray, testing only what's near it: the statics in the grid cells
/// it crosses (nearest first, stopping past the nearest hit), big statics,
/// the bodies that were moving at the last step, slots added since, and
/// `touched` (statics moved since). Before the first step, or with too much
/// changed, it tests everything.
pub fn raycast_near(
    ents: &[Option<Entity>],
    broad: &Broad,
    touched: &[u32],
    o: Vec3,
    dir: Vec3,
    max: f32,
    ignore: Option<u32>,
) -> Option<RayHit> {
    let since = broad.len.min(ents.len());
    if broad.len == 0 || touched.len() > 256 || ents.len() - since > 256 {
        return raycast(ents, o, dir, max, ignore);
    }
    let mut best = None;
    for &i in broad.big.iter().chain(&broad.movers) {
        ray_test(ents, i, o, dir, max, ignore, &mut best);
    }
    for i in since..ents.len() {
        ray_test(ents, i, o, dir, max, ignore, &mut best);
    }
    for &i in touched {
        ray_test(ents, i as usize, o, dir, max, ignore, &mut best);
    }
    // Walk the cells under the ray in x, z (a 2D DDA).
    let c = Broad::CELL;
    let (mut cx, mut cz) = ((o.x / c).floor() as i32, (o.z / c).floor() as i32);
    let (sx, sz) = (if dir.x > 0.0 { 1 } else { -1 }, if dir.z > 0.0 { 1 } else { -1 });
    let first = |p: f32, d: f32, cell: i32| {
        if d.abs() < 1e-9 { f32::INFINITY } else { ((cell + (d > 0.0) as i32) as f32 * c - p) / d }
    };
    let every = |d: f32| if d.abs() < 1e-9 { f32::INFINITY } else { c / d.abs() };
    let (mut tx, mut tz, dtx, dtz) = (first(o.x, dir.x, cx), first(o.z, dir.z, cz), every(dir.x), every(dir.z));
    loop {
        if let Some(v) = broad.cells.get(&(cx, cz)) {
            for &i in v {
                ray_test(ents, i, o, dir, max, ignore, &mut best);
            }
        }
        // A static hit is in a cell it covers: nothing in a cell entered
        // after the nearest hit can be nearer.
        let reach = best.as_ref().map_or(max, |h: &RayHit| h.dist);
        let t_next = tx.min(tz);
        if !t_next.is_finite() || t_next > reach {
            break;
        }
        if tx < tz {
            tx += dtx;
            cx += sx;
        } else {
            tz += dtz;
            cz += sz;
        }
    }
    best
}

fn ray_test(ents: &[Option<Entity>], i: usize, o: Vec3, dir: Vec3, max: f32, ignore: Option<u32>, best: &mut Option<RayHit>) {
    let Some(Some(e)) = ents.get(i) else { return };
    let Some(b) = &e.body else { return };
    if b.sensor || Some(i as u32) == ignore {
        return;
    }
    // Bounding-sphere reject before the exact test; rejects nearly everything.
    let oc = e.pos - o;
    let (r, t) = (b.half.len(), oc.dot(dir));
    let reach = best.as_ref().map_or(max, |h| h.dist);
    if t + r < 0.0 || t - r > reach || oc.dot(oc) - t * t > r * r {
        return;
    }
    let inv = e.rot.conj();
    let (lo, ld) = (inv.rotate(o - e.pos), inv.rotate(dir));
    let (mut t0, mut t1, mut axis_hit) = (0.0f32, max, usize::MAX);
    let mut ok = true;
    for ax in 0..3 {
        let (od, dd, hh) = (lo.get(ax), ld.get(ax), b.half.get(ax));
        if dd.abs() < 1e-8 {
            if od.abs() > hh {
                ok = false;
                break;
            }
            continue;
        }
        let (mut ta, mut tb) = ((-hh - od) / dd, (hh - od) / dd);
        if ta > tb {
            std::mem::swap(&mut ta, &mut tb);
        }
        if ta > t0 {
            t0 = ta;
            axis_hit = ax;
        }
        t1 = t1.min(tb);
        if t0 > t1 {
            ok = false;
            break;
        }
    }
    if !ok || axis_hit == usize::MAX || best.as_ref().is_some_and(|h| h.dist <= t0) {
        return; // miss, or ray starts inside the box
    }
    let mut n = Vec3::ZERO;
    n.set(axis_hit, -ld.get(axis_hit).signum());
    *best = Some(RayHit { id: i as u32, dist: t0, point: o + dir * t0, normal: e.rot.rotate(n) });
}
```

```rs
//! OpenGL via miniquad. Per frame:
//!   1. shadow pass: sun-space depth, packed into an RGBA8 texture (portable
//!      to any GL2/GLES2 driver), covering a box in front of the camera;
//!   2. sky: a full-screen gradient with sun/moon disc and night stars;
//!   3. lit pass: sun with 3x3 PCF shadows, hemisphere ambient, up to 16
//!      point/spot lights, emissive glow, distance fog (opaque, then sorted alpha);
//!   4. HUD quads.

use crate::engine::Engine;
use crate::math::{Mat4, Vec3, v3};
use miniquad::*;

pub const MAX_LIGHTS: usize = 16;
const SHADOW_RES: u32 = 2048;
const SHADOW_SIZE: f32 = 170.0; // meters covered by the shadow map
const SHADOW_DEPTH: f32 = 420.0;

const PRECISION: &str = "#ifdef GL_FRAGMENT_PRECISION_HIGH\nprecision highp float;\n#else\nprecision mediump float;\n#endif\n";

const VS: &str = r#"
attribute vec3 in_pos;
attribute vec4 in_normal; // bytes: 127 is zero, 254 is one
attribute vec4 in_color;  // bytes: rgb, and emissive in a
uniform mat4 mvp;
uniform mat4 model;
uniform vec4 inv_scale2;
varying vec3 v_color;
varying vec3 v_normal;
varying vec3 v_world;
varying float v_emissive;
void main() {
    v_world = (model * vec4(in_pos, 1.0)).xyz;
    v_normal = (model * vec4((in_normal.xyz / 127.0 - 1.0) * inv_scale2.xyz, 0.0)).xyz;
    v_color = in_color.rgb / 255.0;
    v_emissive = in_color.a / 255.0;
    gl_Position = mvp * vec4(in_pos, 1.0);
}"#;

const FS: &str = r#"
varying vec3 v_color;
varying vec3 v_normal;
varying vec3 v_world;
varying float v_emissive;
uniform mat4 light_vp;
uniform vec4 tint;
uniform vec4 extra; // x = entity glow
uniform vec4 eye;
uniform vec4 fog;
uniform vec4 fog_color;
uniform vec4 sun;
uniform vec4 sun_color;
uniform vec4 ambient;
uniform vec4 ground;
uniform vec4 params; // glow, light count, shadows on, shadow texel
uniform vec4 lpos[16]; // xyz, radius
uniform vec4 lcol[16]; // rgb, intensity
uniform vec4 ldir[16]; // xyz, cos(cone) or -1
uniform sampler2D shadow_map;

float unpack(vec4 c) { return dot(c, vec4(1.0, 1.0 / 255.0, 1.0 / 65025.0, 1.0 / 16581375.0)); }

float shadow(float ndl) {
    if (params.z < 0.5 || ndl <= 0.0) return 1.0;
    vec3 p = (light_vp * vec4(v_world, 1.0)).xyz * 0.5 + 0.5;
    if (p.x < 0.0 || p.x > 1.0 || p.y < 0.0 || p.y > 1.0 || p.z > 1.0) return 1.0;
    float bias = 0.0005 + 0.002 * (1.0 - ndl);
    float lit = 0.0;
    for (int x = -1; x <= 1; x++) {
        for (int y = -1; y <= 1; y++) {
            lit += step(p.z - bias, unpack(texture2D(shadow_map, p.xy + vec2(float(x), float(y)) * params.w)));
        }
    }
    return lit / 9.0;
}

void main() {
    vec3 n = normalize(v_normal);
    vec3 base = v_color * tint.rgb;
    float ndl = max(dot(n, sun.xyz), 0.0);
    vec3 light = mix(ground.rgb, ambient.rgb, 0.5 + 0.5 * n.y) + sun_color.rgb * ndl * shadow(ndl);
    for (int i = 0; i < 16; i++) {
        if (float(i) >= params.y) break;
        vec3 l = lpos[i].xyz - v_world;
        float d = length(l);
        l /= max(d, 0.001);
        float att = clamp(1.0 - d / lpos[i].w, 0.0, 1.0);
        float spot = ldir[i].w < -0.5 ? 1.0 : smoothstep(ldir[i].w, ldir[i].w + 0.08, dot(-l, ldir[i].xyz));
        light += lcol[i].rgb * (lcol[i].w * att * att * spot * (0.25 + 0.75 * max(dot(n, l), 0.0)));
    }
    float glow = v_emissive * params.x + extra.x;
    vec3 col = base * light + base * glow * 1.8;
    float d = length(v_world - eye.xyz);
    float f = clamp((d - fog.x) / (fog.y - fog.x), 0.0, 1.0);
    f = f * f * (1.0 - 0.7 * glow);
    gl_FragColor = vec4(mix(col, fog_color.rgb, f), tint.a);
}"#;

const SHADOW_VS: &str = r#"
attribute vec3 in_pos;
uniform mat4 mvp;
void main() { gl_Position = mvp * vec4(in_pos, 1.0); }"#;

const SHADOW_FS: &str = r#"
void main() {
    vec4 e = fract(gl_FragCoord.z * vec4(1.0, 255.0, 65025.0, 16581375.0));
    gl_FragColor = e - e.yzww * vec4(1.0 / 255.0, 1.0 / 255.0, 1.0 / 255.0, 0.0);
}"#;

const SKY_VS: &str = r#"
attribute vec2 in_pos;
varying vec2 v_ndc;
void main() { v_ndc = in_pos; gl_Position = vec4(in_pos, 1.0, 1.0); }"#;

const SKY_FS: &str = r#"
varying vec2 v_ndc;
uniform vec4 cam_fwd;
uniform vec4 cam_right; // scaled by tan(fov/2) * aspect
uniform vec4 cam_up;    // scaled by tan(fov/2)
uniform vec4 horizon;
uniform vec4 top;
uniform vec4 sun;
uniform vec4 sun_color; // a = glow
void main() {
    vec3 dir = normalize(cam_fwd.xyz + cam_right.xyz * v_ndc.x + cam_up.xyz * v_ndc.y);
    float h = clamp(dir.y, 0.0, 1.0);
    vec3 col = mix(horizon.rgb, top.rgb, pow(h, 0.55));
    float s = max(dot(dir, sun.xyz), 0.0);
    float disc = smoothstep(0.9993, 0.9997, s);
    col += sun_color.rgb * (disc * 3.0 + pow(s, 48.0) * 0.4) * step(0.0, sun.y);
    vec3 cell = floor(dir * 260.0);
    float star = fract(sin(dot(cell, vec3(12.9898, 78.233, 37.719))) * 43758.5453);
    col += vec3(step(0.9975, star) * sun_color.a * h * 0.9);
    gl_FragColor = vec4(col, 1.0);
}"#;

const HUD_VS: &str = r#"
attribute vec2 in_pos;
attribute vec4 in_color;
uniform vec4 screen;
varying vec4 v_color;
void main() {
    gl_Position = vec4(in_pos.x / screen.x * 2.0 - 1.0, 1.0 - in_pos.y / screen.y * 2.0, 0.0, 1.0);
    v_color = in_color;
}"#;

const HUD_FS: &str = r#"
varying vec4 v_color;
void main() { gl_FragColor = v_color; }"#;

fn blend() -> Option<BlendState> {
    Some(BlendState::new(Equation::Add, BlendFactor::Value(BlendValue::SourceAlpha), BlendFactor::OneMinusValue(BlendValue::SourceAlpha)))
}

fn shader(ctx: &mut Box<dyn RenderingBackend>, vs: &str, fs: &str, uniforms: &[(&str, UniformType, usize)], images: &[&str]) -> ShaderId {
    let vertex = format!("#version 100\n{vs}");
    let fragment = format!("#version 100\n{PRECISION}{fs}");
    let meta = ShaderMeta {
        images: images.iter().map(|s| s.to_string()).collect(),
        uniforms: UniformBlockLayout { uniforms: uniforms.iter().map(|(n, t, c)| UniformDesc::new(n, *t).array(*c)).collect() },
    };
    ctx.new_shader(ShaderSource::Glsl { vertex: &vertex, fragment: &fragment }, meta).unwrap_or_else(|e| panic!("shader: {e:?}"))
}

fn v4(v: Vec3, w: f32) -> [f32; 4] {
    [v.x, v.y, v.z, w]
}

fn c4(c: [f32; 3], w: f32) -> [f32; 4] {
    [c[0], c[1], c[2], w]
}

    fn sync(&mut self, eng: &Engine) {
        if eng.epoch != self.epoch {
            for m in self.meshes.drain(..).flatten() {
                self.ctx.delete_buffer(m.bind.vertex_buffers[0]);
                self.ctx.delete_buffer(m.bind.index_buffer);
            }
            self.epoch = eng.epoch;
        }
        self.meshes.resize_with(eng.meshes.len(), || None);
        for (i, m) in eng.meshes.iter().enumerate() {
            let rev = eng.mesh_revs[i];
            if self.meshes[i].as_ref().is_some_and(|g| g.rev == rev) {
                continue;
            }
            if let Some(old) = self.meshes[i].take() {
                self.ctx.delete_buffer(old.bind.vertex_buffers[0]);
                self.ctx.delete_buffer(old.bind.index_buffer);
            }
            if m.indices.is_empty() {
                continue;
            }
            let vb = self.ctx.new_buffer(BufferType::VertexBuffer, BufferUsage::Immutable, BufferSource::slice(&m.verts));
            let ib = self.ctx.new_buffer(BufferType::IndexBuffer, BufferUsage::Immutable, BufferSource::slice(&m.indices));
            let radius = m.verts.iter().map(|v| (v.pos[0].powi(2) + v.pos[1].powi(2) + v.pos[2].powi(2)).sqrt()).fold(0.0, f32::max);
            self.meshes[i] = Some(GpuMesh {
                bind: Bindings { vertex_buffers: vec![vb], index_buffer: ib, images: vec![self.shadow_tex] },
                count: m.indices.len() as i32,
                radius,
                rev,
            });
            self.uploaded.push(i);
        }
    }

    /// Meshes uploaded since the last call, whose CPU copies the engine may drop.
    pub fn take_uploaded(&mut self) -> Vec<usize> {
        std::mem::take(&mut self.uploaded)
    }

    /// Orthographic sun view covering a box in front of the camera, snapped
    /// to whole shadow texels so edges don't shimmer as the camera moves.
    fn shadow_matrix(eng: &Engine) -> (Mat4, Vec3) {
        let cam = &eng.camera;
        let sun = eng.env.sun;
        let mut fwd = cam.target - cam.eye;
        fwd.y = 0.0;
        let mut focus = cam.eye + fwd.norm() * (SHADOW_SIZE * 0.3);
        focus.y = 0.0;
        let up = if sun.y.abs() > 0.99 { v3(0., 0., 1.) } else { Vec3::UP };
        let r = (-sun).cross(up).norm();
        let u = r.cross(-sun);
        let texel = SHADOW_SIZE / SHADOW_RES as f32;
        let snap = |a: f32| (a / texel).floor() * texel - a;
        focus += r * snap(focus.dot(r)) + u * snap(focus.dot(u));
        let view = Mat4::look_at(focus + sun * (SHADOW_DEPTH * 0.5), focus, up);
        let h = SHADOW_SIZE * 0.5;
        (Mat4::ortho(-h, h, -h, h, 0.0, SHADOW_DEPTH) * view, focus)
    }

    pub fn draw(&mut self, eng: &Engine) {
        self.sync(eng);
        let (w, h) = eng.screen;
        let env = &eng.env;
        let cam = &eng.camera;
        let aspect = w / h.max(1.0);
        // The camera keeps 0.4 m off walls, so a 0.25 m near plane clips
        // nothing and buys depth precision for flat details far away.
        let proj = Mat4::perspective(cam.fov, aspect, 0.25, env.fog_end + 50.0);
        let view = Mat4::look_at(cam.eye, cam.target, Vec3::UP);
        let vp = proj * view;
        let fwd = (cam.target - cam.eye).norm();
        let plane = eng.mesh_ids.get("plane").copied();

        // Cull, then split into opaque and back-to-front alpha lists.
        let mut opaque = Vec::new();
        let mut alpha = Vec::new();
        let mut casters = Vec::new();
        let (light_vp, focus) = Self::shadow_matrix(eng);
        let shadows = env.shadows && env.sun.y > 0.05;
        for e in eng.ents.iter().flatten() {
            let (Some(mi), true) = (e.mesh, e.visible) else { continue };
            let Some(Some(g)) = self.meshes.get(mi) else { continue };
            let r = g.radius * e.scale.x.abs().max(e.scale.y.abs()).max(e.scale.z.abs());
            let to = e.pos - cam.eye;
            let dist = to.len();
            if shadows && e.color[3] >= 1.0 && Some(mi) != plane {
                let d = e.pos - focus;
                if (d.x * d.x + d.z * d.z).sqrt() - r < SHADOW_SIZE * 0.75 {
                    casters.push((e, mi));
                }
            }
            if dist - r > env.fog_end || to.dot(fwd) < -r {
                continue;
            }
            if e.color[3] < 1.0 { alpha.push((dist, e, mi)) } else { opaque.push((dist, e, mi)) }
        }
        alpha.sort_by(|a, b| b.0.total_cmp(&a.0));
        self.draw_calls = 0;

        // 1. Shadow map.
        if shadows {
            self.ctx
                .begin_pass(Some(self.shadow_pass), PassAction::Clear { color: Some((1., 1., 1., 1.)), depth: Some(1.0), stencil: None });
            self.ctx.apply_pipeline(&self.shadow);
            for (e, mi) in &casters {
                let g = self.meshes[*mi].as_ref().unwrap();
                let mvp = light_vp * Mat4::trs_q(e.pos, e.rot, e.scale);
                self.ctx.apply_bindings(&g.bind);
                self.ctx.apply_uniforms(UniformsSource::table(&mvp.0));
                self.ctx.draw(0, g.count, 1);
                self.draw_calls += 1;
            }
            self.ctx.end_render_pass();
        }

        // 2. Sky.
        self.ctx.begin_default_pass(PassAction::Clear {
            color: Some((env.sky[0], env.sky[1], env.sky[2], 1.0)),
            depth: Some(1.0),
            stencil: None,
        });
        let t = (cam.fov * 0.5).tan();
        let right = fwd.cross(Vec3::UP).norm();
        let up = right.cross(fwd);
        let sky = SkyUniforms {
            cam_fwd: v4(fwd, 0.),
            cam_right: v4(right * (t * aspect), 0.),
            cam_up: v4(up * t, 0.),
            horizon: c4(env.sky, 1.),
            top: c4(env.sky_top, 1.),
            sun: v4(env.sun, 0.),
            sun_color: c4(env.sun_color, env.glow),
        };
        self.ctx.apply_pipeline(&self.sky);
        self.ctx.apply_bindings(&self.sky_bind);
        self.ctx.apply_uniforms(UniformsSource::table(&sky));
        self.ctx.draw(0, 3, 1);

        // 3. The world, lit by the nearest lights.
        let mut uni = Uniforms {
            mvp: [0.0; 16],
            model: [0.0; 16],
            light_vp: light_vp.0,
            inv_scale2: [1.0; 4],
            tint: [1.0; 4],
            extra: [0.0; 4],
            eye: v4(cam.eye, 0.),
            fog: [env.fog_start, env.fog_end, 0.0, 0.0],
            fog_color: c4(env.sky, 1.),
            sun: v4(env.sun, 0.),
            sun_color: c4(env.sun_color, 1.),
            ambient: c4(env.ambient, 1.),
            ground: c4(env.ground, 1.),
            params: [env.glow, 0.0, if shadows { 1.0 } else { 0.0 }, 1.0 / SHADOW_RES as f32],
            lpos: [[0.0; 4]; MAX_LIGHTS],
            lcol: [[0.0; 4]; MAX_LIGHTS],
            ldir: [[0.0; 4]; MAX_LIGHTS],
        };
        self.lights_used = 0;
        if env.glow > 0.01 {
            let mut near: Vec<(f32, Vec3, Vec3, &crate::engine::Light)> = eng
                .lights
                .iter()
                .flatten()
                .filter(|l| l.intensity > 0.0)
                .filter_map(|l| eng.light_world(l).map(|(p, d)| ((p - cam.eye).len() - l.radius, p, d, l)))
                .filter(|(score, ..)| *score < env.fog_end)
                .collect();
            near.sort_by(|a, b| a.0.total_cmp(&b.0));
            for (k, (_, p, d, l)) in near.iter().take(MAX_LIGHTS).enumerate() {
                uni.lpos[k] = v4(*p, l.radius);
                uni.lcol[k] = c4(l.color, l.intensity * env.glow);
                uni.ldir[k] = v4(*d, l.cone);
                self.lights_used += 1;
            }
            uni.params[1] = self.lights_used as f32;
        }
        for (pipe, list) in [(&self.opaque, &opaque), (&self.alpha, &alpha)] {
            if list.is_empty() {
                continue;
            }
            self.ctx.apply_pipeline(pipe);
            for (_, e, mi) in list.iter() {
                let g = self.meshes[*mi].as_ref().unwrap();
                let model = Mat4::trs_q(e.pos, e.rot, e.scale);
                uni.mvp = (vp * model).0;
                uni.model = model.0;
                let s = e.scale;
                uni.inv_scale2 = [1.0 / (s.x * s.x), 1.0 / (s.y * s.y), 1.0 / (s.z * s.z), 0.0];
                uni.tint = e.color;
                uni.extra[0] = e.glow;
                self.ctx.apply_bindings(&g.bind);
                self.ctx.apply_uniforms(UniformsSource::table(&uni));
                self.ctx.draw(0, g.count, 1);
                self.draw_calls += 1;
            }
        }
        self.draw_hud(eng);
        self.ctx.end_render_pass();
    }

    fn draw_hud(&mut self, eng: &Engine) {
        let verts = eng.hud.len() / 6;
        if verts == 0 {
            return;
        }
        if self.hud_bind.as_ref().is_none_or(|(_, cap)| *cap < verts) {
            if let Some((b, _)) = self.hud_bind.take() {
                self.ctx.delete_buffer(b.vertex_buffers[0]);
                self.ctx.delete_buffer(b.index_buffer);
            }
            let cap = verts.next_power_of_two().max(4096);
            let vb = self.ctx.new_buffer(BufferType::VertexBuffer, BufferUsage::Stream, BufferSource::empty::<f32>(cap * 6));
            let idx: Vec<u32> = (0..cap as u32).collect();
            let ib = self.ctx.new_buffer(BufferType::IndexBuffer, BufferUsage::Immutable, BufferSource::slice(&idx));
            self.hud_bind = Some((Bindings { vertex_buffers: vec![vb], index_buffer: ib, images: vec![] }, cap));
        }
        let (bind, _) = self.hud_bind.as_ref().unwrap();
        self.ctx.buffer_update(bind.vertex_buffers[0], BufferSource::slice(&eng.hud));
        self.ctx.apply_pipeline(&self.hud);
        self.ctx.apply_bindings(bind);
        let screen = [eng.screen.0, eng.screen.1, 0.0, 0.0];
        self.ctx.apply_uniforms(UniformsSource::table(&screen));
        self.ctx.draw(0, verts as i32, 1);
    }

    pub fn commit(&mut self) {
        self.ctx.commit_frame();
    }

    /// Read the back buffer and write a binary PPM.
    pub fn screenshot(&mut self, path: &str, w: i32, h: i32) -> std::io::Result<()> {
        let mut px = vec![0u8; (w * h * 4) as usize];
        unsafe {
            miniquad::gl::glReadPixels(0, 0, w, h, miniquad::gl::GL_RGBA, miniquad::gl::GL_UNSIGNED_BYTE, px.as_mut_ptr() as *mut _);
        }
        let mut out = format!("P6\n{w} {h}\n255\n").into_bytes();
        for y in (0..h).rev() {
            for x in 0..w {
                let i = ((y * w + x) * 4) as usize;
                out.extend_from_slice(&px[i..i + 3]);
            }
        }
        std::fs::write(path, out)
    }
}
```

```rs
local traceback = debug.traceback
-- Handler lists are copy-on-write so (un)subscribing during dispatch is safe.
function engine.on(ev, fn)
  local t = {table.unpack(handlers[ev] or {})}
  t[#t + 1] = fn
  handlers[ev] = t
  return fn
end
function engine.off(ev, fn)
  local t = {}
  for _, f in ipairs(handlers[ev] or {}) do if f ~= fn then t[#t + 1] = f end end
  handlers[ev] = t
end
-- Pausing stops ticks, and key events arrive as "menu_key" instead of "key"
-- until resumed. Keys held at the moment of pausing are released to the game
-- first, so a held trigger or accelerator doesn't stick through the menu.
local set_paused = engine._pause
engine._pause = nil
function engine.pause(on)
  for _, k in ipairs(set_paused(on == true)) do engine.emit("key", k, false) end
end
function engine.once(ev, fn)
  local function w(...) engine.off(ev, w); return fn(...) end
  return engine.on(ev, w)
end
function engine.emit(ev, ...)
  local t = handlers[ev]
  if not t then return end
  for i = 1, #t do
    local ok, err = xpcall(t[i], traceback, ...)
    if not ok then engine._error(ev, tostring(err)) end
  end

pub struct Script {
    pub lua: Lua,
    emit: Function,

impl Script {
    /// Fresh Lua state, API bound, `main.lua` executed.
    pub fn load(eng: &Shared) -> Result<Script> {
        // Game scripts are trusted local code; `debug` gives us tracebacks.
        let lua = unsafe { Lua::unsafe_new_with(StdLib::ALL_SAFE | StdLib::DEBUG, LuaOptions::default()) };
        let root = eng.borrow().root.clone();
        let api = bind(&lua, eng)?;
        lua.globals().set("engine", api)?;
        lua.globals().set("__ROOT", root.to_string_lossy().replace('\\', "/"))?;
        lua.load(KERNEL).set_name("=kernel").exec()?;
        let pkg: Table = lua.globals().get("package")?;
        let r = root.display();
        pkg.set("path", format!("{r}/?.lua;{r}/?/init.lua"))?;
        let main = root.join("main.lua");
        let src = std::fs::read_to_string(&main).map_err(|e| mlua::Error::external(format!("{}: {e}", main.display())))?;
        lua.load(src).set_name("@main.lua").exec()?;
        let emit = lua.globals().get::<Table>("engine")?.get::<Function>("emit")?;
        Ok(Script { lua, emit })
    }

    /// Emit an event. Handler errors are caught inside the bus, so this only
    /// fails if the bus itself is broken.
    pub fn emit(&self, eng: &Shared, ev: &str, args: impl IntoLuaMulti) {
        let mut a = match args.into_lua_multi(&self.lua) {
            Ok(a) => a,
            Err(e) => return report(eng, ev, &e.to_string()),
        };
        a.push_front(Value::String(self.lua.create_string(ev).unwrap()));
        if let Err(e) = self.emit.call::<()>(a) {
            report(eng, ev, &e.to_string());
        }
    }
}

/// Resolve a script-supplied path under the game root. Absolute paths and

fn bind(lua: &Lua, eng: &Shared) -> Result<Table> {
    let api = lua.create_table()?;
    macro_rules! f {
        ($name:literal, |$e:ident, $lua:ident, $args:tt : $ty:ty| $body:expr) => {{
            let eng = eng.clone();
            api.set(
                $name,
                lua.create_function(move |$lua, $args: $ty| {
                    #[allow(unused_mut)]
                    let mut $e = eng.borrow_mut();
                    $body
                })?,
            )?;
        }};
    }

    // ---- entities -------------------------------------------------------
    f!("spawn", |e, _lua, t: Table| {
        let size = vec3(&t.get("size")?, Vec3::ONE)?;
        let mut ent = Entity::new(vec3(&t.get("pos")?, Vec3::ZERO)?);
        let angle = |k: &str| -> Result<f32> { Ok(t.get::<Option<f32>>(k)?.unwrap_or(0.0)) };
        ent.rot = Quat::euler(angle("yaw")?, angle("pitch")?, angle("roll")?);
        let mesh: Option<String> = t.get("mesh")?;
        ent.mesh = match &mesh {
            Some(m) => Some(mesh_id(&e, m)?),
            None => None,
        };
        // Built-in unit primitives scale to `size`; custom meshes default to 1.
        let unit = mesh.as_deref().is_none_or(mesh::builtins_contains);
        ent.scale = vec3(&t.get("scale")?, if unit { size } else { Vec3::ONE })?;
        ent.color = color(&t.get("color")?)?;
        ent.tag = t.get::<Option<String>>("tag")?.unwrap_or_default();
        ent.visible = t.get::<Option<bool>>("visible")?.unwrap_or(true);
        ent.glow = t.get::<Option<f32>>("glow")?.unwrap_or(0.0);
        if let Some(kind) = t.get::<Option<String>>("body")? {
            let k = match kind.as_str() {
                "static" => Kind::Static,
                "dynamic" => Kind::Dynamic,
                "kinematic" => Kind::Kinematic,
                o => return Err(mlua::Error::runtime(format!("unknown body kind '{o}'"))),
            };
            let mut b = Body::new(k, size * 0.5);
            b.lock_rotation = t.get::<Option<bool>>("lock_rotation")?.unwrap_or(false);
            b.set_mass(t.get::<Option<f32>>("mass")?.unwrap_or(1.0));
            b.friction = t.get::<Option<f32>>("friction")?.unwrap_or(b.friction);
            b.restitution = t.get::<Option<f32>>("restitution")?.unwrap_or(b.restitution);
            b.gravity = t.get::<Option<f32>>("gravity")?.unwrap_or(b.gravity);
            b.sensor = t.get::<Option<bool>>("sensor")?.unwrap_or(false);
            b.step = t.get::<Option<f32>>("step")?.unwrap_or(0.0);
            ent.body = Some(b);
        }
        Ok(e.spawn(ent))
    });
    f!("despawn", |e, _lua, id: u32| Ok(e.despawn(id)));
    f!("exists", |e, _lua, id: u32| Ok(e.get(id).is_some()));
    f!("pos", |e, lua, id: u32| multi(lua, e.get(id).map(|x| (x.pos.x, x.pos.y, x.pos.z))));
    f!("set_pos", |e, _lua, (id, x, y, z): (u32, f32, f32, f32)| {
        if let Some(en) = e.pose_mut(id) {
            en.pos = v3(x, y, z);
            en.prev = None; // a teleport: don't draw it sliding there
        }
        Ok(())
    });
    f!("yaw", |e, _lua, id: u32| Ok(e.get(id).map(|x| x.rot.yaw())));
    // Turns about the world up axis, keeping any pitch and roll.
    f!("set_yaw", |e, _lua, (id, y): (u32, f32)| {
        if let Some(en) = e.pose_mut(id) {
            en.rot = (Quat::axis_angle(Vec3::UP, y - en.rot.yaw()) * en.rot).norm();
        }
        Ok(())
    });
    f!("orient", |e, _lua, (id, yaw, pitch, roll): (u32, f32, Option<f32>, Option<f32>)| {
        if let Some(en) = e.pose_mut(id) {
            en.rot = Quat::euler(yaw, pitch.unwrap_or(0.0), roll.unwrap_or(0.0));
        }
        Ok(())
    });
    f!("rot", |e, lua, id: u32| multi(lua, e.get(id).map(|x| (x.rot.x, x.rot.y, x.rot.z, x.rot.w))));
    f!("set_rot", |e, _lua, (id, x, y, z, w): (u32, f32, f32, f32, f32)| {
        if let Some(en) = e.pose_mut(id) {
            en.rot = Quat { x, y, z, w }.norm();
        }
        Ok(())
    });
    // Local axes in world space. Forward is +Z, up is +Y, right is -X.
    for (name, local) in [("forward", v3(0., 0., 1.)), ("up", Vec3::UP), ("right", v3(-1., 0., 0.))] {
        let eng = eng.clone();
        api.set(
            name,
            lua.create_function(move |lua, id: u32| {
                multi(lua, eng.borrow().get(id).map(|x| x.rot.rotate(local)).map(|v| (v.x, v.y, v.z)))
            })?,
        )?;
    }
    f!("vel", |e, lua, id: u32| multi(lua, e.get(id).and_then(|x| x.body.as_ref()).map(|b| (b.vel.x, b.vel.y, b.vel.z))));
    f!("set_vel", |e, _lua, (id, x, y, z): (u32, f32, f32, f32)| {
        if let Some(b) = e.get_mut(id).and_then(|en| en.body.as_mut()) {
            b.vel = v3(x, y, z);
        }
        Ok(())
    });
    f!("angvel", |e, lua, id: u32| multi(lua, e.get(id).and_then(|x| x.body.as_ref()).map(|b| (b.ang_vel.x, b.ang_vel.y, b.ang_vel.z))));
    f!("set_angvel", |e, _lua, (id, x, y, z): (u32, f32, f32, f32)| {
        if let Some(b) = e.get_mut(id).and_then(|en| en.body.as_mut()) {
            b.ang_vel = v3(x, y, z);
        }
        Ok(())
    });
    // Impulse, optionally at a world point (which also adds spin).
    f!("push", |e, _lua, (id, x, y, z, px, py, pz): (u32, f32, f32, f32, Option<f32>, Option<f32>, Option<f32>)| {
        if let Some(en) = e.get_mut(id) {
            let (pos, rot) = (en.pos, en.rot);
            if let Some(b) = en.body.as_mut() {
                let imp = v3(x, y, z);
                b.vel += imp * b.inv_mass;
                if let (Some(px), Some(py), Some(pz)) = (px, py, pz) {
                    let r = v3(px, py, pz) - pos;
                    let local = rot.conj().rotate(r.cross(imp)).mul(b.inv_inertia);
                    b.ang_vel += rot.rotate(local);
                }
            }
        }
        Ok(())
    });
    f!("grounded", |e, _lua, id: u32| Ok(e.get(id).and_then(|x| x.body.as_ref()).is_some_and(|b| b.grounded)));
    f!("set_color", |e, _lua, (id, r, g, b, a): (u32, f32, f32, f32, Option<f32>)| {
        if let Some(en) = e.get_mut(id) {
            en.color = [r, g, b, a.unwrap_or(1.0)];
        }
        Ok(())
    });
    f!("set_glow", |e, _lua, (id, g): (u32, f32)| {
        if let Some(en) = e.get_mut(id) {
            en.glow = g;
        }
        Ok(())
    });
    f!("set_visible", |e, _lua, (id, v): (u32, bool)| {
        if let Some(en) = e.get_mut(id) {
            en.visible = v;
        }
        Ok(())
    });
    f!("set_scale", |e, _lua, (id, s): (u32, Value)| {
        let s = vec3(&s, Vec3::ONE)?;
        if let Some(en) = e.get_mut(id) {
            en.scale = s;
        }
        Ok(())
    });
    f!("tag", |e, _lua, id: u32| Ok(e.get(id).map(|x| x.tag.clone())));
    // Swap an entity's model, e.g. a respray. Its body and scale stay as they are.
    f!("set_mesh", |e, _lua, (id, name): (u32, String)| {
        let m = mesh_id(&e, &name)?;
        if let Some(en) = e.get_mut(id) {
            en.mesh = Some(m);
        }
        Ok(())
    });
    f!("mesh_of", |e, _lua, id: u32| {
        let m = e.get(id).and_then(|x| x.mesh);
        Ok(m.and_then(|m| e.mesh_ids.iter().find(|(_, i)| **i == m).map(|(n, _)| n.clone())))
    });
    f!("find", |e, _lua, tag: String| Ok(e.find(&tag)));
    f!("nearest", |e, lua, (x, y, z, r, tag, ignore): (f32, f32, f32, f32, Option<String>, Option<u32>)| {
        let p = v3(x, y, z);
        let best = e
            .ents
            .iter()
            .enumerate()
            .filter_map(|(i, en)| Some((i as u32, en.as_ref()?)))
            .filter(|(i, en)| Some(*i) != ignore && tag.as_ref().is_none_or(|t| &en.tag == t))
            .map(|(i, en)| (i, (en.pos - p).len()))
            .filter(|&(_, d)| d <= r)
            .min_by(|a, b| a.1.total_cmp(&b.1));
        multi(lua, best)
    });
    f!("raycast", |e, lua, (ox, oy, oz, dx, dy, dz, max, ignore): (f32, f32, f32, f32, f32, f32, Option<f32>, Option<u32>)| {
        let dir = v3(dx, dy, dz).norm();
        match e.raycast(v3(ox, oy, oz), dir, max.unwrap_or(1000.0), ignore) {
            Some(h) => (h.id, h.dist, h.point.x, h.point.y, h.point.z, h.normal.x, h.normal.y, h.normal.z).into_lua_multi(lua),
            None => Ok(MultiValue::new()),
        }
    });

    // The ground as a heightfield: {x0, z0, step, w, d, heights = {w * d
    // numbers, row by row toward +z}, colors = {r, g, b per height}?, roads =
    // {ax, az, ay, bx, bz, by, half width, road number, asphalt half width,
    // side 0 raised (0 or 1), side 1 raised, ...}?, kerb = meters?}; nil
    // removes it.
    f!("terrain", |e, _lua, t: Option<Table>| {
        let Some(t) = t else {
            e.set_terrain(None, None);
            return Ok(());
        };
        let (w, d): (usize, usize) = (t.get("w")?, t.get("d")?);
        let step: f32 = t.get("step")?;
        if w < 2 || d < 2 || step <= 0.0 {
            return Err(mlua::Error::runtime("terrain needs w, d >= 2 and step > 0"));
        }
        // Heights as numbers, or a file: w * d little-endian i16 centimeters,
        // then optionally a byte per point picking its color from `palette`.
        let (heights, classes): (Vec<f32>, Option<Vec<u8>>) = match t.get::<Value>("heights")? {
            Value::String(rel) => {
                let path = game_path(&e.root, &rel.to_str()?)?;
                let bytes = std::fs::read(&path).map_err(|err| mlua::Error::runtime(format!("{}: {err}", path.display())))?;
                let n = w * d;
                if bytes.len() != 2 * n && bytes.len() != 3 * n {
                    return Err(mlua::Error::runtime(format!("{} is {} bytes, not {} or {}", rel.to_str()?, bytes.len(), 2 * n, 3 * n)));
                }
                let hs = bytes[..2 * n].chunks(2).map(|b| i16::from_le_bytes([b[0], b[1]]) as f32 / 100.0).collect();
                (hs, (bytes.len() == 3 * n).then(|| bytes[2 * n..].to_vec()))
            }
            Value::Table(tb) => (tb.sequence_values().collect::<Result<_>>()?, None),
            _ => return Err(mlua::Error::runtime("terrain heights must be a table of numbers or a file name")),
        };
        if heights.len() != w * d {
            return Err(mlua::Error::runtime(format!("terrain has {} heights, not w * d = {}", heights.len(), w * d)));
        }
        let palette = match t.get::<Option<Table>>("palette")? {
            None => None,
            Some(p) => Some(
                p.sequence_values::<Table>().map(|c| color(&Value::Table(c?)).map(|c| [c[0], c[1], c[2]])).collect::<Result<Vec<_>>>()?,
            ),
        };
        let colors = match t.get::<Option<Table>>("colors")? {
            None => match (classes, palette) {
                (Some(cl), Some(pal)) if !pal.is_empty() => Some(cl.iter().map(|&k| pal[(k as usize).min(pal.len() - 1)]).collect()),
                _ => None,
            },
            Some(c) => {
                let v: Vec<f32> = c.sequence_values().collect::<Result<_>>()?;
                if v.len() != 3 * w * d {
                    return Err(mlua::Error::runtime(format!("terrain has {} color values, not 3 * w * d = {}", v.len(), 3 * w * d)));
                }
                Some(v.chunks(3).map(|c| [c[0], c[1], c[2]]).collect())
            }
        };
        // Roads over it, level across, raised past the asphalt on a side
        // with a pavement: {ax, az, ay, bx, bz, by, half width, which road,
        // asphalt half width, side 0 raised, side 1 raised, ...}.
        let roads = match t.get::<Option<Table>>("roads")? {
            None => Vec::new(),
            Some(r) => {
                let v: Vec<f32> = r.sequence_values().collect::<Result<_>>()?;
                if !v.len().is_multiple_of(11) {
                    return Err(mlua::Error::runtime(
                        "terrain roads come in elevens: ax, az, ay, bx, bz, by, half width, road, asphalt half width, raised, raised",
                    ));
                }
                v.chunks(11)
                    .map(|c| RoadSeg {
                        a: v3(c[0], c[2], c[1]),
                        b: v3(c[3], c[5], c[4]),
                        half: c[6],
                        road: c[7] as u32,
                        carriage: c[8],
                        raised: [c[9] != 0.0, c[10] != 0.0],
                    })
                    .collect()
            }
        };
        let mut terrain = Terrain::new(t.get("x0")?, t.get("z0")?, step, w, d, heights);
        terrain.kerb = t.get::<Option<f32>>("kerb")?.unwrap_or(0.0);
        let terrain = terrain.with_roads(roads);
        e.set_terrain(Some(terrain), colors);
        Ok(())
    });
    // Ground height and normal at (x, z); nothing off the heightfield.
    f!("terrain_height", |e, lua, (x, z): (f32, f32)| {
        multi(lua, e.terrain.as_ref().and_then(|t| t.surface(x, z)).map(|(h, n)| (h, n.x, n.y, n.z)))
    });

    // ---- meshes ---------------------------------------------------------
    f!("mesh", |e, _lua, (name, parts): (String, Table)| {
        let mut out = Mesh::default();
        for p in parts.sequence_values::<Table>() {
            let p = p?;
            let part = p.get::<Option<String>>("mesh")?.unwrap_or("cube".into());
            let src = mesh_id(&e, &part)?;
            if e.gpu_only.contains(&src) {
                return Err(mlua::Error::runtime(format!("mesh '{part}' is drawn but too big to build other meshes from")));
            }
            let c = color(&p.get("color")?)?;
            let size = vec3(&p.get("size")?, Vec3::ONE)?;
            let mut src = e.meshes[src].clone();
            if let Some(axis) = p.get::<Option<String>>("axis")? {
                src.reorient(&axis).map_err(mlua::Error::runtime)?;
            }
            let angle = |k: &str| -> Result<f32> { Ok(p.get::<Option<f32>>(k)?.unwrap_or(0.0)) };
            let turn = (angle("yaw")?, angle("pitch")?, angle("roll")?);
            let glow = p.get::<Option<f32>>("emissive")?.unwrap_or(0.0);
            out.append_tilted(&src, vec3(&p.get("pos")?, Vec3::ZERO)?, turn, size, v3(c[0], c[1], c[2]), glow);
        }
        let tris = out.tri_count();
        e.add_mesh(&name, out);
        Ok(tris)
    });
    f!("load_obj", |e, _lua, (name, path): (String, String)| {
        let m = mesh::load_obj(&game_path(&e.root, &path)?).map_err(mlua::Error::runtime)?;
        let tris = m.tri_count();
        e.add_mesh(&name, m);
        Ok(tris)
    });

    // ---- audio ----------------------------------------------------------
    f!("synth", |e, _lua, (name, t): (String, Table)| {
        let d = Synth::default();
        let num = |k: &str, def: f32| -> Result<f32> { Ok(t.get::<Option<f32>>(k)?.unwrap_or(def)) };
        let wave = match t.get::<Option<String>>("wave")?.as_deref().unwrap_or("square") {
            "sine" => Wave::Sine,
            "square" => Wave::Square,
            "saw" => Wave::Saw,
            "triangle" => Wave::Triangle,
            "noise" => Wave::Noise,
            o => return Err(mlua::Error::runtime(format!("unknown wave '{o}'"))),
        };
        let notes = match t.get::<Option<Table>>("notes")? {
            Some(n) => n.sequence_values::<f32>().collect::<Result<Vec<_>>>()?,
            None => Vec::new(),
        };
        let s = Synth {
            wave,
            freq: num("freq", d.freq)?,
            freq_end: num("freq_end", 0.0)?,
            dur: num("dur", d.dur)?,
            attack: num("attack", d.attack)?,
            release: num("release", d.release)?,
            volume: num("volume", d.volume)?,
            lowpass: num("lowpass", 1.0)?,
            notes,
            note_len: num("note_len", d.note_len)?,
            looped: t.get::<Option<bool>>("loop")?.unwrap_or(false),
        };
        let buf = s.render();
        let secs = buf.len() as f32 / audio::RATE as f32;
        e.sounds.insert(name, std::sync::Arc::new(buf));
        Ok(secs)
    });
    f!("load_wav", |e, _lua, (name, path): (String, String)| {
        let full = game_path(&e.root, &path)?;
        let bytes = std::fs::read(&full).map_err(|err| mlua::Error::runtime(format!("{}: {err}", full.display())))?;
        let buf = audio::load_wav(&bytes).map_err(|err| mlua::Error::runtime(format!("{path}: {err}")))?;
        let secs = buf.len() as f32 / audio::RATE as f32;
        e.sounds.insert(name, std::sync::Arc::new(buf));
        Ok(secs)
    });
    f!("play", |e, _lua, (name, t): (String, Option<Table>)| {
        let data = e.sounds.get(&name).cloned().ok_or_else(|| mlua::Error::runtime(format!("unknown sound '{name}'")))?;
        let get = |k: &str| -> Result<Option<f32>> { t.as_ref().map_or(Ok(None), |t| t.get::<Option<f32>>(k)) };
        let (volume, pitch) = (get("volume")?.unwrap_or(1.0), get("pitch")?.unwrap_or(1.0));
        let looped = t.as_ref().map_or(Ok(None), |t| t.get::<Option<bool>>("loop"))?.unwrap_or(false);
        let follow = t.as_ref().map_or(Ok(None), |t| t.get::<Option<u32>>("follow"))?;
        let mut world = match t.as_ref().map_or(Ok(Value::Nil), |t| t.get::<Value>("pos"))? {
            Value::Nil => None,
            v => Some(vec3(&v, Vec3::ZERO)?),
        };
        if let Some(f) = follow {
            world = e.get(f).map(|x| x.pos);
        }
        let ui = t.as_ref().map_or(Ok(None), |t| t.get::<Option<bool>>("ui"))?.unwrap_or(false);
        let id = e.mixer.lock().unwrap().play(data, volume, pitch, looped, world, ui);
        if let Some(f) = follow {
            e.follow.push((id, f));
        }
        Ok(id)
    });
    f!("set_voice", |e, _lua, (id, t): (u32, Table)| {
        let world = match t.get::<Value>("pos")? {
            Value::Nil => None,
            v => Some(vec3(&v, Vec3::ZERO)?),
        };
        e.mixer.lock().unwrap().set(id, t.get("volume")?, t.get("pitch")?, world);
        Ok(())
    });
    f!("stop", |e, _lua, id: u32| Ok(e.mixer.lock().unwrap().stop(id)));
    f!("playing", |e, _lua, id: u32| Ok(e.mixer.lock().unwrap().playing(id)));
    f!("volume", |e, _lua, v: Option<f32>| {
        let mut m = e.mixer.lock().unwrap();
        if let Some(v) = v {
            m.master = v.clamp(0.0, 4.0);
        }
        Ok(m.master)
    });

    // ---- lights -------------------------------------------------------------
    fn apply_light(l: &mut crate::engine::Light, t: &Table) -> Result<()> {
        l.pos = vec3(&t.get("pos")?, l.pos)?;
        l.dir = vec3(&t.get("dir")?, l.dir)?.norm();
        if let Value::Table(_) = t.get::<Value>("color")? {
            let c = color(&t.get("color")?)?;
            l.color = [c[0], c[1], c[2]];
        }
        l.intensity = t.get::<Option<f32>>("intensity")?.unwrap_or(l.intensity);
        l.radius = t.get::<Option<f32>>("radius")?.unwrap_or(l.radius);
        if let Some(deg) = t.get::<Option<f32>>("cone")? {
            l.cone = if deg >= 180.0 { -1.0 } else { deg.to_radians().cos() };
        }
        if let Some(f) = t.get::<Option<u32>>("follow")? {
            l.follow = Some(f);
        }
        Ok(())
    }
    f!("light", |e, _lua, t: Table| {
        let mut l = crate::engine::Light {
            pos: Vec3::ZERO,
            dir: v3(0., -1., 0.),
            color: [1.0, 0.9, 0.7],
            intensity: 1.0,
            radius: 12.0,
            cone: -1.0,
            follow: None,
        };
        apply_light(&mut l, &t)?;
        e.lights.push(Some(l));
        Ok(e.lights.len() as u32 - 1)
    });
    f!("set_light", |e, _lua, (id, t): (u32, Table)| {
        if let Some(Some(l)) = e.lights.get_mut(id as usize) {
            apply_light(l, &t)?;
        }
        Ok(())
    });
    f!("remove_light", |e, _lua, id: u32| {
        if let Some(l) = e.lights.get_mut(id as usize) {
            *l = None;
        }
        Ok(())
    });
    f!("lighting", |e, _lua, t: Table| {
        let rgb = |k: &str, cur: [f32; 3]| -> Result<[f32; 3]> {
            Ok(match t.get::<Value>(k)? {
                Value::Table(_) => {
                    let c = color(&t.get(k)?)?;
                    [c[0], c[1], c[2]]
                }
                _ => cur,
            })
        };
        let env = &mut e.env;
        env.sky = rgb("sky", env.sky)?;
        env.sky_top = rgb("sky_top", env.sky_top)?;
        env.sun_color = rgb("sun_color", env.sun_color)?;
        env.ambient = rgb("ambient", env.ambient)?;
        env.ground = rgb("ground", env.ground)?;
        if let Value::Table(_) = t.get::<Value>("sun")? {
            env.sun = vec3(&t.get("sun")?, env.sun)?.norm();
        }
        env.glow = t.get::<Option<f32>>("glow")?.unwrap_or(env.glow).clamp(0.0, 1.0);
        env.shadows = t.get::<Option<bool>>("shadows")?.unwrap_or(env.shadows);
        Ok(())
    });

    // ---- camera & environment ------------------------------------------
    f!("camera", |e, _lua, (ex, ey, ez, tx, ty, tz, fov): (f32, f32, f32, f32, f32, f32, Option<f32>)| {
        e.camera.eye = v3(ex, ey, ez);
        e.camera.target = v3(tx, ty, tz);
        if let Some(f) = fov {
            e.camera.fov = f.to_radians();
        }
        Ok(())
    });
    f!("camera_pos", |e, _lua, (): ()| Ok((e.camera.eye.x, e.camera.eye.y, e.camera.eye.z)));
    f!("sky", |e, _lua, (r, g, b): (f32, f32, f32)| Ok(e.env.sky = [r, g, b]));
    f!("fog", |e, _lua, (s, en): (f32, f32)| {
        e.env.fog_start = s;
        e.env.fog_end = en;
        Ok(())
    });
    f!("sun", |e, _lua, (x, y, z): (f32, f32, f32)| Ok(e.env.sun = v3(x, y, z).norm()));
    f!("gravity", |e, _lua, g: f32| Ok(e.env.gravity = g));

    // ---- input ----------------------------------------------------------
    f!("key", |e, _lua, k: String| Ok(e.keys.contains(&k)));
    f!("mouse", |e, _lua, (): ()| Ok(e.mouse_delta));
    f!("grab_mouse", |e, _lua, g: bool| Ok(e.mouse_grab = g));
    f!("cursor", |e, _lua, (): ()| Ok(e.cursor));
    // The first gamepad: sticks -1..1 (up and right positive), triggers 0..1. Nil without one.
    f!("pad", |e, lua, (): ()| {
        let Some(p) = &e.pad else { return Ok(Value::Nil) };
        let t = lua.create_table()?;
        t.set("name", p.name.as_str())?;
        for (k, v) in [("lx", p.lx), ("ly", p.ly), ("rx", p.rx), ("ry", p.ry), ("lt", p.lt), ("rt", p.rt)] {
            t.set(k, v)?;
        }
        Ok(Value::Table(t))
    });
    f!("_pause", |e, _lua, on: bool| Ok(e.set_paused(on)));
    f!("paused", |e, _lua, (): ()| Ok(e.paused));
    f!("fullscreen", |e, _lua, on: Option<bool>| {
        if let Some(on) = on {
            e.fullscreen = on;
        }
        Ok(e.fullscreen)
    });

    // ---- HUD ------------------------------------------------------------
    f!("screen", |e, _lua, (): ()| Ok(e.screen));
    f!("rect", |e, _lua, (x, y, w, h, r, g, b, a): (f32, f32, f32, f32, f32, f32, f32, Option<f32>)| {
        Ok(e.hud_rect(x, y, w, h, [r, g, b, a.unwrap_or(1.0)]))
    });
    f!(
        "text",
        |e, _lua, (s, x, y, size, r, g, b, a): (String, f32, f32, Option<f32>, Option<f32>, Option<f32>, Option<f32>, Option<f32>)| {

        t.set("voices", e.mixer.lock().unwrap().voice_count())?;

        api.set(
            name,
            lua.create_function(move |_lua, dir: String| {
                let path = game_path(&eng.borrow().root, &dir)?;
                let mut names: Vec<String> = std::fs::read_dir(path)
                    .map(|rd| {
                        rd.filter_map(|d| {
                            let d = d.ok()?;
                            (d.file_type().ok()?.is_dir() == want_dir).then(|| d.file_name().into_string().ok())?
                        })
                        .collect()
                    })
                    .unwrap_or_default();
                names.sort();
                Ok(names)
            })?,
        )?;

    // One plain value per key. Bad keys or unsavable values are script bugs
    // and raise; disk trouble returns nil, message so the game can carry on.
    f!("save", |e, _lua, (key, v): (String, Value)| {
        save::check_key(&key).map_err(mlua::Error::runtime)?;

            Value::Nil => None,
            v => Some(save::encode(&v).map_err(|err| mlua::Error::runtime(format!("save '{key}': {err}")))?),
        };
        let keys = e.saves.keys();
        if text.is_some() && !keys.contains(&key) {
            if keys.len() >= save::MAX_KEYS {
                return Err(mlua::Error::runtime(format!("save '{key}': already {} saved keys", save::MAX_KEYS)));
            }
            // Mod keys are "mod.<name>.<key>"; one mod gets a quarter of the room.
            if let Some(m) = key.strip_prefix("mod.").and_then(|k| k.split_once('.')).map(|(m, _)| format!("mod.{m}."))
                && keys.iter().filter(|k| k.starts_with(&m)).count() >= save::MAX_KEYS / 4
            {
                return Err(mlua::Error::runtime(format!("save '{key}': this mod already has {} saved keys", save::MAX_KEYS / 4)));
            }
        }
        Ok(match e.saves.write(&key, text.as_deref()) {
            Ok(()) => (true, None),
            Err(err) => (false, Some(format!("save '{key}': {err}"))),
        })
    });
    f!("load", |e, lua, key: String| {
        save::check_key(&key).map_err(mlua::Error::runtime)?;
        let text = match e.saves.read(&key) {
            Ok(Some(t)) => t,
            Ok(None) => return Ok((Value::Nil, None)),
            Err(err) => return Ok((Value::Nil, Some(format!("load '{key}': {err}")))),
        };
        Ok(match save::decode(lua, &text) {
            Ok(v) => (v, None),
            Err(err) => {
                let moved = e.saves.quarantine(&key).map_or(String::new(), |p| format!("; moved it to {}", p.display()));
                (Value::Nil, Some(format!("load '{key}': {err}{moved}")))
            }
        })
    });
    f!("saves", |e, _lua, (): ()| Ok(e.saves.keys()));

    f!("_error", |e, _lua, (ev, msg): (String, String)| {
        eprintln!("[lua:{ev}] {msg}");
        e.error = Some(format!("{ev}: {msg}"));
        Ok(())
    });
    Ok(api)
}
```

```rs
//! Audio: a software mixer with pitch and 3D panning, an sfxr-style synth so
//! games need no sound files, a WAV loader for when they want them, and
//! the smallest output backend per platform (ALSA via dlopen on Linux, so
//! there is no build dependency; cpal elsewhere). No device means silence,
//! never an error.

use crate::math::Vec3;
use std::sync::atomic::{AtomicBool, Ordering};
use std::sync::{Arc, Mutex};

/// Rate every sound is stored at. The mixer resamples to the device.
pub const RATE: u32 = 44100;

pub type Samples = Arc<Vec<f32>>;

struct Voice {
    id: u32,
    data: Samples,
    pos: f64,
    pitch: f32,
    volume: f32,
    looped: bool,
    world: Option<Vec3>,
    gains: (f32, f32),
    /// Interface sounds keep playing while the game is paused.
    ui: bool,
}

pub struct Mixer {
    voices: Vec<Voice>,
    next: u32,
    pub out_rate: u32,
    pub listener: Vec3,
    pub listener_right: Vec3,
    pub master: f32,
    /// Freezes every voice except `ui` ones, where they are.
    pub paused: bool,
}

impl Default for Mixer {
    fn default() -> Mixer {
        Mixer { voices: Vec::new(), next: 1, out_rate: RATE, listener: Vec3::ZERO, listener_right: Vec3::ZERO, master: 0.7, paused: false }
    }
}

/// Distance rolloff and equal-power pan for a voice at `world`.
fn spatial(world: Option<Vec3>, listener: Vec3, right: Vec3) -> (f32, f32) {
    let Some(p) = world else { return (1.0, 1.0) };
    let d = p - listener;
    let dist = d.len();
    let att = 1.0 / (1.0 + (dist / 12.0).powi(2));
    let pan = if dist > 0.01 { d.norm().dot(right).clamp(-1.0, 1.0) * 0.8 } else { 0.0 };
    let a = (pan + 1.0) * std::f32::consts::FRAC_PI_4;
    (att * a.cos() * std::f32::consts::SQRT_2, att * a.sin() * std::f32::consts::SQRT_2)
}

impl Mixer {
    pub fn play(&mut self, data: Samples, volume: f32, pitch: f32, looped: bool, world: Option<Vec3>, ui: bool) -> u32 {
        let id = self.next;
        self.next += 1;
        let gains = spatial(world, self.listener, self.listener_right);
        self.voices.push(Voice { id, data, pos: 0.0, pitch, volume, looped, world, gains, ui });
        id
    }

    pub fn set(&mut self, id: u32, volume: Option<f32>, pitch: Option<f32>, world: Option<Vec3>) {
        if let Some(v) = self.voices.iter_mut().find(|v| v.id == id) {
            v.volume = volume.unwrap_or(v.volume);
            v.pitch = pitch.unwrap_or(v.pitch);
            v.world = world.or(v.world);
        }
    }

    pub fn stop(&mut self, id: u32) {
        self.voices.retain(|v| v.id != id);
    }

    pub fn stop_all(&mut self) {
        self.voices.clear();
        self.paused = false;
    }

    pub fn playing(&self, id: u32) -> bool {
        self.voices.iter().any(|v| v.id == id)
    }

    pub fn voice_count(&self) -> usize {
        self.voices.len()
    }

    /// Mix into interleaved stereo `out` (overwrites it).
    pub fn mix(&mut self, out: &mut [f32]) {
        out.fill(0.0);
        let frames = out.len() / 2;
        let ratio = RATE as f64 / self.out_rate as f64;
        let (listener, right) = (self.listener, self.listener_right);
        for v in &mut self.voices {
            if self.paused && !v.ui {
                continue;
            }
            let n = v.data.len();
            if n == 0 {
                v.looped = false;
                v.pos = 0.0;
                continue;
            }
            // Ramp gains across the buffer so moving sources don't click.
            let (gl0, gr0) = v.gains;
            let (gl1, gr1) = spatial(v.world, listener, right);
            v.gains = (gl1, gr1);
            let step = v.pitch.max(0.0) as f64 * ratio;
            for f in 0..frames {
                if v.pos >= n as f64 {
                    if !v.looped {
                        break;
                    }
                    v.pos %= n as f64;
                }
                let i = v.pos as usize;
                let frac = (v.pos - i as f64) as f32;
                let next = if i + 1 < n {
                    v.data[i + 1]
                } else if v.looped {
                    v.data[0]
                } else {
                    0.0
                };
                let s = (v.data[i] + (next - v.data[i]) * frac) * v.volume;
                let t = f as f32 / frames as f32;
                out[2 * f] += s * (gl0 + (gl1 - gl0) * t);
                out[2 * f + 1] += s * (gr0 + (gr1 - gr0) * t);
                v.pos += step;
            }
        }
        self.voices.retain(|v| v.looped || v.pos < v.data.len() as f64);
        for s in out.iter_mut() {
            *s = (*s * self.master).clamp(-1.0, 1.0);
        }
    }
}

#[derive(Clone, Copy, PartialEq, Debug)]
pub enum Wave {
    Sine,
    Square,
    Saw,
    Triangle,
    Noise,
}

pub struct Synth {
    pub wave: Wave,
    pub freq: f32,
    pub freq_end: f32,
    pub dur: f32,
    pub attack: f32,
    pub release: f32,
    pub volume: f32,
    /// One-pole lowpass amount: 1 = off, smaller = darker.
    pub lowpass: f32,
    /// If non-empty, play these frequencies in sequence, `note_len` each.
    pub notes: Vec<f32>,
    pub note_len: f32,
    /// Make the buffer loop seamlessly (whole periods, no envelope).
    pub looped: bool,
}

impl Default for Synth {
    fn default() -> Synth {
        Synth {
            wave: Wave::Square,
            freq: 440.0,
            freq_end: 0.0,
            dur: 0.3,
            attack: 0.005,
            release: 0.05,
            volume: 0.5,
            lowpass: 1.0,
            notes: Vec::new(),
            note_len: 0.12,
            looped: false,
        }
    }
}

impl Synth {
    pub fn render(&self) -> Vec<f32> {
        let sr = RATE as f32;
        let segments: Vec<(f32, f32, f32)> = if self.notes.is_empty() {
            let end = if self.freq_end > 0.0 { self.freq_end } else { self.freq };
            vec![(self.freq, end, self.dur)]
        } else {
            self.notes.iter().map(|&f| (f, f, self.note_len)).collect()
        };
        let mut out = Vec::new();
        let mut rng: u32 = 0x9E37_79B9;
        let mut noise = 0.0f32;
        let mut lp = 0.0f32;
        let a = self.lowpass.clamp(0.001, 1.0).powi(2);
        for (f0, f1, dur) in segments {
            let (mut f0, mut dur) = (f0, dur);
            let looped = self.looped && self.wave != Wave::Noise;
            if looped {
                // Whole number of samples per period, whole number of periods.
                let period = (sr / f0).round().max(2.0);
                f0 = sr / period;
                dur = ((dur * f0).round().max(1.0)) / f0;
            }
            let n = (dur * sr).round() as usize;
            let mut phase = 0.0f32;
            for k in 0..n {
                let t = k as f32 / n as f32;
                let f = if looped { f0 } else { f0 * (f1 / f0).powf(t) };
                let prev = phase;
                phase = (phase + f / sr).fract();
                let x = match self.wave {
                    Wave::Sine => (phase * std::f32::consts::TAU).sin(),
                    Wave::Square => {
                        if phase < 0.5 {
                            1.0
                        } else {
                            -1.0
                        }
                    }
                    Wave::Saw => 2.0 * phase - 1.0,
                    Wave::Triangle => 1.0 - 4.0 * (phase - 0.5).abs(),
                    Wave::Noise => {
                        // Sample-and-hold white noise: lower freq = rumblier.
                        if phase < prev || k == 0 {
                            rng ^= rng << 13;
                            rng ^= rng >> 17;
                            rng ^= rng << 5;
                            noise = (rng as f32 / u32::MAX as f32) * 2.0 - 1.0;
                        }
                        noise
                    }
                };
                let secs = k as f32 / sr;
                let env = if looped {
                    1.0
                } else {
                    let att = if self.attack > 0.0 { (secs / self.attack).min(1.0) } else { 1.0 };
                    let rel = if self.release > 0.0 { ((dur - secs) / self.release).clamp(0.0, 1.0) } else { 1.0 };
                    att * rel
                };
                lp += a * (x - lp);
                out.push(lp * env * self.volume);
            }
        }
        out
    }
}

/// 8/16-bit PCM WAV, any channel count and rate, to mono at `RATE`.
pub fn load_wav(bytes: &[u8]) -> Result<Vec<f32>, String> {
    let u16le = |b: &[u8], o: usize| u16::from_le_bytes([b[o], b[o + 1]]);
    let u32le = |b: &[u8], o: usize| u32::from_le_bytes([b[o], b[o + 1], b[o + 2], b[o + 3]]);
    if bytes.len() < 12 || &bytes[0..4] != b"RIFF" || &bytes[8..12] != b"WAVE" {
        return Err("not a RIFF/WAVE file".into());
    }
    let (mut fmt, mut data) = (None, None);
    let mut o = 12;
    while o + 8 <= bytes.len() {
        let size = u32le(bytes, o + 4) as usize;
        let body = &bytes[o + 8..(o + 8 + size).min(bytes.len())];
        match &bytes[o..o + 4] {
            b"fmt " if body.len() >= 16 => fmt = Some((u16le(body, 0), u16le(body, 2), u32le(body, 4), u16le(body, 14))),
            b"data" => data = Some(body),
            _ => {}
        }
        o += 8 + size + (size & 1);
    }
    let (format, channels, rate, bits) = fmt.ok_or("no fmt chunk")?;
    let data = data.ok_or("no data chunk")?;
    if format != 1 || !(bits == 8 || bits == 16) || channels == 0 {
        return Err(format!("unsupported WAV (format {format}, {bits}-bit); use 8/16-bit PCM"));
    }
    let width = bits as usize / 8 * channels as usize;
    let mono: Vec<f32> = data
        .chunks_exact(width)
        .map(|fr| {
            let ch = (0..channels as usize).map(|c| {
                if bits == 8 { (fr[c] as f32 - 128.0) / 128.0 } else { i16::from_le_bytes([fr[2 * c], fr[2 * c + 1]]) as f32 / 32768.0 }
            });
            ch.sum::<f32>() / channels as f32
        })
        .collect();
    if rate == RATE || mono.is_empty() {
        return Ok(mono);
    }
    let ratio = rate as f64 / RATE as f64;
    let n = (mono.len() as f64 / ratio) as usize;
    Ok((0..n)
        .map(|k| {
            let p = k as f64 * ratio;
            let i = p as usize;
            let f = (p - i as f64) as f32;
            let b = mono.get(i + 1).copied().unwrap_or(mono[i]);
            mono[i] + (b - mono[i]) * f
        })
        .collect())
}

/// Keeps the output device alive; dropping it stops playback.
pub struct Output {
    stop: Arc<AtomicBool>,
    #[allow(dead_code)]
    stream: Option<Box<dyn std::any::Any>>,
    thread: Option<std::thread::JoinHandle<()>>,
}

impl Drop for Output {
    fn drop(&mut self) {
        self.stop.store(true, Ordering::Relaxed);
        if let Some(t) = self.thread.take() {
            let _ = t.join();
        }
    }
}

pub fn start(mixer: Arc<Mutex<Mixer>>) -> Option<Output> {
    let stop = Arc::new(AtomicBool::new(false));
    match backend::start(mixer, stop.clone()) {
        Ok((thread, stream)) => Some(Output { stop, stream, thread }),
        Err(e) => {
            eprintln!("audio: {e} (running silent)");
            None
        }
    }
}

#[cfg(target_os = "linux")]
mod backend {
    //! ALSA through dlopen: works wherever libasound.so.2 exists (including
    //! PulseAudio/PipeWire setups), and needs nothing at build time.
    use super::{Mixer, RATE};
    use std::ffi::{CString, c_char, c_int, c_long, c_uint, c_ulong, c_void};
    use std::sync::atomic::{AtomicBool, Ordering};
    use std::sync::{Arc, Mutex};

    unsafe extern "C" {
        fn dlopen(file: *const c_char, flag: c_int) -> *mut c_void;
        fn dlsym(handle: *mut c_void, name: *const c_char) -> *mut c_void;
    }
    type Open = unsafe extern "C" fn(*mut *mut c_void, *const c_char, c_int, c_int) -> c_int;
    type SetParams = unsafe extern "C" fn(*mut c_void, c_int, c_int, c_uint, c_uint, c_int, c_uint) -> c_int;
    type Writei = unsafe extern "C" fn(*mut c_void, *const c_void, c_ulong) -> c_long;
    type Recover = unsafe extern "C" fn(*mut c_void, c_int, c_int) -> c_int;
    type Close = unsafe extern "C" fn(*mut c_void) -> c_int;

    struct Pcm(*mut c_void);
    unsafe impl Send for Pcm {}

    const FRAMES: usize = 512;

    pub type Started = (Option<std::thread::JoinHandle<()>>, Option<Box<dyn std::any::Any>>);

    pub fn start(mixer: Arc<Mutex<Mixer>>, stop: Arc<AtomicBool>) -> Result<Started, String> {
        unsafe {
            let lib = dlopen(c"libasound.so.2".as_ptr(), 2 /* RTLD_NOW */);
            if lib.is_null() {
                return Err("libasound.so.2 not found".into());
            }
            let sym = |name: &str| {
                let c = CString::new(name).unwrap();
                let p = dlsym(lib, c.as_ptr());
                if p.is_null() { Err(format!("missing {name}")) } else { Ok(p) }
            };
            let open: Open = std::mem::transmute(sym("snd_pcm_open")?);
            let set_params: SetParams = std::mem::transmute(sym("snd_pcm_set_params")?);
            let writei: Writei = std::mem::transmute(sym("snd_pcm_writei")?);
            let recover: Recover = std::mem::transmute(sym("snd_pcm_recover")?);
            let close: Close = std::mem::transmute(sym("snd_pcm_close")?);

            let mut pcm = std::ptr::null_mut();
            if open(&mut pcm, c"default".as_ptr(), 0 /* playback */, 0) < 0 {
                return Err("no ALSA output device".into());
            }
            // S16_LE, interleaved, stereo, resampling allowed, 40 ms latency.
            if set_params(pcm, 2, 3, 2, RATE, 1, 40_000) < 0 {
                close(pcm);
                return Err("ALSA rejected 44.1 kHz stereo".into());
            }
            mixer.lock().unwrap().out_rate = RATE;
            let pcm = Pcm(pcm);
            let thread = std::thread::spawn(move || {
                let pcm = pcm;
                let mut f = vec![0.0f32; FRAMES * 2];
                let mut s = vec![0i16; FRAMES * 2];
                while !stop.load(Ordering::Relaxed) {
                    mixer.lock().unwrap().mix(&mut f);
                    for (d, x) in s.iter_mut().zip(&f) {
                        *d = (x * 32767.0) as i16;
                    }
                    let r = writei(pcm.0, s.as_ptr() as *const c_void, FRAMES as c_ulong);
                    if r < 0 {
                        recover(pcm.0, r as c_int, 1);
                    }
                }
                close(pcm.0);
            });
            Ok((Some(thread), None))
        }
    }
}

#[cfg(not(target_os = "linux"))]
mod backend {
    use super::Mixer;
    use cpal::traits::{DeviceTrait, HostTrait, StreamTrait};
    use std::sync::atomic::AtomicBool;
    use std::sync::{Arc, Mutex};

    pub type Started = (Option<std::thread::JoinHandle<()>>, Option<Box<dyn std::any::Any>>);

    pub fn start(mixer: Arc<Mutex<Mixer>>, _stop: Arc<AtomicBool>) -> Result<Started, String> {
        let device = cpal::default_host().default_output_device().ok_or("no output device")?;
        let config = device.default_output_config().map_err(|e| e.to_string())?;
        if config.sample_format() != cpal::SampleFormat::F32 {
            return Err(format!("unsupported sample format {:?}", config.sample_format()));
        }
        let channels = config.channels() as usize;
        mixer.lock().unwrap().out_rate = config.sample_rate().0;
        let mut buf = Vec::new();
        let stream = device
            .build_output_stream(
                &config.into(),
                move |data: &mut [f32], _| {
                    let frames = data.len() / channels;
                    buf.resize(frames * 2, 0.0);
                    mixer.lock().unwrap().mix(&mut buf);
                    for (f, out) in data.chunks_mut(channels).enumerate() {
                        for (c, s) in out.iter_mut().enumerate() {
                            *s = buf[2 * f + (c & 1)];
                        }
                    }
                },
                |e| eprintln!("audio: {e}"),
                None,
            )
            .map_err(|e| e.to_string())?;
        stream.play().map_err(|e| e.to_string())?;
        Ok((None, Some(Box::new(stream))))
    }
}
```

```rs
//! Gamepads. Buttons become ordinary key names (`pad_a`, `pad_start`,
//! `pad_up`, ...) so they travel the same road as the keyboard; sticks and
//! triggers are read with `engine.pad()`.
//!
//! On Linux the kernel's evdev interface is read directly (no udev, so still
//! no build dependencies); every driver that follows the kernel gamepad API
//! (xpad, hid-playstation, hid-nintendo, ...) reports the same codes. On
//! Windows and macOS, gilrs does the work.

/// The first connected pad, normalized: sticks -1..1 with up and right
/// positive, triggers 0..1.
#[derive(Clone, Debug, Default, PartialEq)]
pub struct State {
    pub name: String,
    pub lx: f32,
    pub ly: f32,
    pub rx: f32,
    pub ry: f32,
    pub lt: f32,
    pub rt: f32,
}

/// What happened since the last poll.
#[derive(Clone, Debug, PartialEq)]
pub enum Event {
    Connected(String),
    Disconnected(String),
    Button(&'static str, bool),
}

/// Triggers also act as buttons once pressed past this.
const TRIGGER_DOWN: f32 = 0.5;

/// Track analog triggers crossing the button threshold.
fn trigger_edge(name: &'static str, old: f32, new: f32, out: &mut Vec<Event>) {
    if (old >= TRIGGER_DOWN) != (new >= TRIGGER_DOWN) {
        out.push(Event::Button(name, new >= TRIGGER_DOWN));
    }
}

mod evdev {
    use super::{Event, State, trigger_edge};
    use std::fs::File;
    use std::io::Read;
    use std::os::fd::AsRawFd;

    unsafe extern "C" {
        fn ioctl(fd: i32, request: std::ffi::c_ulong, ...) -> i32;
    }

    // <linux/input.h>
    const EV_KEY: u16 = 0x01;
    const EV_ABS: u16 = 0x03;
    const BTN_GAMEPAD: u16 = 0x130;
    const ABS_X: u16 = 0x00;
    const ABS_Y: u16 = 0x01;
    const ABS_Z: u16 = 0x02;
    const ABS_RX: u16 = 0x03;
    const ABS_RY: u16 = 0x04;
    const ABS_RZ: u16 = 0x05;
    const ABS_GAS: u16 = 0x09;
    const ABS_BRAKE: u16 = 0x0a;
    const ABS_HAT0X: u16 = 0x10;
    const ABS_HAT0Y: u16 = 0x11;
    const ABS_COUNT: usize = 0x40;

    /// struct input_event: a timeval, then type, code and value.
    const EVENT_SIZE: usize = 2 * std::mem::size_of::<std::ffi::c_long>() + 8;

    const fn ioc_read(nr: u32, size: usize) -> std::ffi::c_ulong {
        ((2 << 30) | ((size as u32) << 16) | ((b'E' as u32) << 8) | nr) as std::ffi::c_ulong
    }
    pub(super) const fn eviocgname(len: usize) -> std::ffi::c_ulong {
        ioc_read(0x06, len)
    }
    pub(super) const fn eviocgbit(ev: u16, len: usize) -> std::ffi::c_ulong {
        ioc_read(0x20 + ev as u32, len)
    }
    pub(super) const fn eviocgabs(abs: u16) -> std::ffi::c_ulong {
        ioc_read(0x40 + abs as u32, 24)
    }

    pub(super) fn button(code: u16) -> Option<&'static str> {
        Some(match code {
            0x130 => "pad_a",
            0x131 => "pad_b",
            0x133 => "pad_y",
            0x134 => "pad_x",
            0x136 => "pad_lb",
            0x137 => "pad_rb",
            0x13a => "pad_back",
            0x13b => "pad_start",
            0x13c => "pad_guide",
            0x13d => "pad_l3",
            0x13e => "pad_r3",
            0x220 => "pad_up",
            0x221 => "pad_down",
            0x222 => "pad_left",
            0x223 => "pad_right",
            _ => return None,
        })
    }

    /// An axis's range, from EVIOCGABS: value, min, max, fuzz, flat, resolution.
    #[derive(Clone, Copy, Default)]
    pub(super) struct Range {
        pub min: i32,
        pub max: i32,
        pub flat: i32,
    }

    impl Range {
        /// -1..1 for sticks (with the driver's dead zone), 0..1 for triggers.
        pub fn stick(&self, v: i32) -> f32 {
            let mid = (self.min as f32 + self.max as f32) / 2.0;
            let half = (self.max as f32 - self.min as f32) / 2.0;
            if half <= 0.0 || (v as f32 - mid).abs() <= self.flat as f32 {
                return 0.0;
            }
            ((v as f32 - mid) / half).clamp(-1.0, 1.0)
        }
        pub fn trigger(&self, v: i32) -> f32 {
            let span = (self.max - self.min) as f32;
            if span <= 0.0 { 0.0 } else { ((v - self.min) as f32 / span).clamp(0.0, 1.0) }
        }
    }

    /// One open gamepad and its decoded state.
    pub(super) struct Device {
        file: Option<File>,
        pub path: std::path::PathBuf,
        pub state: State,
        pub ranges: [Range; ABS_COUNT],
        hat: (i32, i32),
        /// The xpad driver (Xbox pads) sends Y as BTN_WEST and X as BTN_NORTH,
        /// the other way round from the gamepad spec other drivers follow.
        pub swap_xy: bool,
    }

    impl Device {
        fn open(path: &std::path::Path) -> Option<Device> {
            use std::os::unix::fs::OpenOptionsExt;
            const O_NONBLOCK: i32 = 0o4000;
            let file = std::fs::OpenOptions::new().read(true).custom_flags(O_NONBLOCK).open(path).ok()?;
            let fd = file.as_raw_fd();
            // A gamepad has the gamepad buttons and at least one stick.
            let mut keys = [0u8; 96];
            let mut abs = [0u8; 8];
            unsafe {
                if ioctl(fd, eviocgbit(EV_KEY, keys.len()), keys.as_mut_ptr()) < 0
                    || ioctl(fd, eviocgbit(EV_ABS, abs.len()), abs.as_mut_ptr()) < 0
                {
                    return None;
                }
            }
            let has = |bits: &[u8], n: u16| bits.get(n as usize / 8).is_some_and(|b| b & (1 << (n % 8)) != 0);
            if !has(&keys, BTN_GAMEPAD) || !has(&abs, ABS_X) {
                return None;
            }
            let mut name = [0u8; 128];
            unsafe { ioctl(fd, eviocgname(name.len()), name.as_mut_ptr()) };
            let end = name.iter().position(|&c| c == 0).unwrap_or(name.len());
            let mut d = Device::new(path, String::from_utf8_lossy(&name[..end]).trim().to_string());
            for code in 0..ABS_COUNT as u16 {
                if has(&abs, code) {
                    let mut info = [0i32; 6];
                    if unsafe { ioctl(fd, eviocgabs(code), info.as_mut_ptr()) } >= 0 {
                        d.ranges[code as usize] = Range { min: info[1], max: info[2], flat: info[4] };
                    }
                }
            }
            d.swap_xy = path
                .file_name()
                .and_then(|n| std::fs::read_link(format!("/sys/class/input/{}/device/device/driver", n.to_string_lossy())).ok())
                .is_some_and(|drv| drv.file_name().is_some_and(|f| f == "xpad"));
            d.file = Some(file);
            Some(d)
        }

        #[cfg(test)]
        pub(super) fn attach(&mut self, file: File) {
            self.file = Some(file);
        }

        pub(super) fn new(path: &std::path::Path, name: String) -> Device {
            Device {
                file: None,
                path: path.to_path_buf(),
                state: State { name, ..Default::default() },
                ranges: [Range::default(); ABS_COUNT],
                hat: (0, 0),
                swap_xy: false,
            }
        }

        /// Apply one input_event.
        pub(super) fn apply(&mut self, ty: u16, code: u16, value: i32, out: &mut Vec<Event>) {
            match ty {
                EV_KEY => {
                    // Digital triggers are covered by the analog ones (or added below).
                    let code = match code {
                        0x133 if self.swap_xy => 0x134,
                        0x134 if self.swap_xy => 0x133,
                        c => c,
                    };
                    if let Some(name) = button(code) {
                        out.push(Event::Button(name, value != 0));
                    } else if code == 0x138 || code == 0x139 {
                        let (name, now) = if code == 0x138 { ("pad_lt", &mut self.state.lt) } else { ("pad_rt", &mut self.state.rt) };
                        let new = if value != 0 { 1.0 } else { 0.0 };
                        let old = std::mem::replace(now, new);
                        trigger_edge(name, old, new, out);
                    }
                }
                EV_ABS => {
                    let r = self.ranges.get(code as usize).copied().unwrap_or_default();
                    let s = &mut self.state;
                    match code {
                        ABS_X => s.lx = r.stick(value),
                        ABS_Y => s.ly = -r.stick(value),
                        ABS_RX => s.rx = r.stick(value),
                        ABS_RY => s.ry = -r.stick(value),
                        ABS_Z | ABS_BRAKE => {
                            let new = r.trigger(value);
                            trigger_edge("pad_lt", std::mem::replace(&mut s.lt, new), new, out);
                        }
                        ABS_RZ | ABS_GAS => {
                            let new = r.trigger(value);
                            trigger_edge("pad_rt", std::mem::replace(&mut s.rt, new), new, out);
                        }
                        ABS_HAT0X | ABS_HAT0Y => {
                            let v = value.signum();
                            let (old, names) = if code == ABS_HAT0X {
                                (std::mem::replace(&mut self.hat.0, v), ("pad_left", "pad_right"))
                            } else {
                                (std::mem::replace(&mut self.hat.1, v), ("pad_up", "pad_down"))
                            };
                            if old != v {
                                if old != 0 {
                                    out.push(Event::Button(if old < 0 { names.0 } else { names.1 }, false));
                                }
                                if v != 0 {
                                    out.push(Event::Button(if v < 0 { names.0 } else { names.1 }, true));
                                }
                            }
                        }
                        _ => {}
                    }
                }
                _ => {}
            }
        }

        /// Read everything pending. False once the device is gone.
        pub(super) fn read(&mut self, out: &mut Vec<Event>) -> bool {
            let mut buf = [0u8; EVENT_SIZE * 64];
            loop {
                let Some(file) = self.file.as_mut() else { return false };
                match file.read(&mut buf) {
                    Ok(0) => return false,
                    Ok(n) => {
                        for ev in buf[..n].as_chunks::<EVENT_SIZE>().0 {
                            let t = EVENT_SIZE - 8;
                            let ty = u16::from_ne_bytes([ev[t], ev[t + 1]]);
                            let code = u16::from_ne_bytes([ev[t + 2], ev[t + 3]]);
                            let value = i32::from_ne_bytes([ev[t + 4], ev[t + 5], ev[t + 6], ev[t + 7]]);
                            self.apply(ty, code, value, out);
                        }
                    }
                    Err(e) if e.kind() == std::io::ErrorKind::WouldBlock => return true,
                    Err(_) => return false,
                }
            }
        }
    }

    pub struct Pads {
        devices: Vec<Device>,
        next_scan: f64,
    }

    impl Pads {
        pub fn new() -> Pads {
            Pads { devices: Vec::new(), next_scan: 0.0 }
        }

        /// Read input, and look for newly plugged-in pads every couple of seconds.
        pub fn poll(&mut self, now: f64, out: &mut Vec<Event>) {
            if now >= self.next_scan {
                self.next_scan = now + 2.0;
                let mut paths: Vec<_> = std::fs::read_dir("/dev/input")
                    .map(|rd| {
                        rd.flatten()
                            .map(|e| e.path())
                            .filter(|p| p.file_name().and_then(|n| n.to_str()).is_some_and(|n| n.starts_with("event")))
                            .collect()
                    })
                    .unwrap_or_default();
                paths.sort();
                for p in paths {
                    if self.devices.iter().any(|d| d.path == p) {
                        continue;
                    }
                    if let Some(d) = Device::open(&p) {
                        out.push(Event::Connected(d.state.name.clone()));
                        self.devices.push(d);
                    }
                }
            }
            self.devices.retain_mut(|d| {
                let alive = d.read(out);
                if !alive {
                    out.push(Event::Disconnected(d.state.name.clone()));
                }
                alive
            });
        }

        pub fn state(&self) -> Option<&State> {
            self.devices.first().map(|d| &d.state)
        }
    }
}

            let old = self.state.take().unwrap_or_default();

                let trig = |b| p.button_data(b).map_or(0.0, |d| d.value());

                    lx: p.value(Axis::LeftStickX),
```

```rs
//! Save data. A script stores a plain Lua value (tables, strings, numbers,
//! booleans) under a key, and each key is one small file of Lua table syntax
//! in the user's data folder. Files are parsed as data, never run as code,
//! so a hand-edited or hostile save can't do anything but fail to load.

use mlua::{Lua, Table, Value};
use std::collections::HashMap;
use std::fmt::Write;
use std::path::PathBuf;

/// Largest encoded value, and most keys a store will hold. Saves are for
/// progress and settings, so these are generous; they bound what a runaway
/// or hostile script can write to disk.
pub const MAX_BYTES: usize = 256 * 1024;
pub const MAX_KEYS: usize = 64;
const MAX_DEPTH: usize = 64;
const HEADER: &str = "-- auto save data: plain Lua table syntax, read as data and never run.\nreturn ";

/// Where saves live: a folder on disk, or (headless runs, screenshots) memory
/// that is gone when the process exits.
pub enum Store {
    Dir(PathBuf),
    Memory(HashMap<String, String>),
}

impl Default for Store {
    fn default() -> Store {
        Store::Memory(HashMap::new())
    }
}

impl Store {
    fn file(dir: &std::path::Path, key: &str) -> PathBuf {
        dir.join(format!("{key}.lua"))
    }

    pub fn read(&self, key: &str) -> std::io::Result<Option<String>> {
        match self {
            Store::Memory(m) => Ok(m.get(key).cloned()),
            Store::Dir(d) => match std::fs::read_to_string(Self::file(d, key)) {
                Ok(s) => Ok(Some(s)),
                Err(e) if e.kind() == std::io::ErrorKind::NotFound => Ok(None),
                Err(e) => Err(e),
            },
        }
    }

    /// Write `text` under `key`, or delete the key when `text` is None. The
    /// file is replaced atomically, so a crash mid-save keeps the old one.
    pub fn write(&mut self, key: &str, text: Option<&str>) -> std::io::Result<()> {
        match (self, text) {
            (Store::Memory(m), Some(t)) => drop(m.insert(key.to_string(), t.to_string())),
            (Store::Memory(m), None) => drop(m.remove(key)),
            (Store::Dir(d), Some(t)) => {
                std::fs::create_dir_all(&*d)?;
                let tmp = d.join(format!("{key}.lua.tmp"));
                std::fs::write(&tmp, t)?;
                std::fs::rename(&tmp, Self::file(d, key))?;
            }
            (Store::Dir(d), None) => match std::fs::remove_file(Self::file(d, key)) {
                Err(e) if e.kind() != std::io::ErrorKind::NotFound => return Err(e),
                _ => {}
            },
        }
        Ok(())
    }

    /// Set aside an unreadable save (as `key.lua.bad`) so the next save
    /// doesn't overwrite what a person may want to fix by hand.
    pub fn quarantine(&mut self, key: &str) -> Option<PathBuf> {
        match self {
            Store::Memory(m) => {
                m.remove(key);
                None
            }
            Store::Dir(d) => {
                let bad = d.join(format!("{key}.lua.bad"));
                std::fs::rename(Self::file(d, key), &bad).ok().map(|_| bad)
            }
        }
    }

    /// Stored keys, sorted.
    pub fn keys(&self) -> Vec<String> {
        let mut k: Vec<String> = match self {
            Store::Memory(m) => m.keys().cloned().collect(),
            Store::Dir(d) => std::fs::read_dir(d)
                .map(|rd| rd.flatten().filter_map(|e| e.file_name().into_string().ok()?.strip_suffix(".lua").map(String::from)).collect())
                .unwrap_or_default(),
        };
        k.sort();
        k
    }
}

/// The per-user folder for a game's saves, e.g. `~/.local/share/auto/game`.
pub fn user_dir(game: &str) -> Option<PathBuf> {
    let env = |k: &str| std::env::var_os(k).filter(|v| !v.is_empty()).map(PathBuf::from);
    let base = if cfg!(windows) {
        env("APPDATA")?
    } else if cfg!(target_os = "macos") {
        env("HOME")?.join("Library/Application Support")
    } else {
        env("XDG_DATA_HOME").or_else(|| Some(env("HOME")?.join(".local/share")))?
    };
    Some(base.join("auto").join(game))
}

/// Keys become file names, so they are short and plain: letters, digits,
/// `_`, `-` and `.`, not starting with a dot, and never a name Windows reserves.
pub fn check_key(key: &str) -> Result<(), String> {
    let plain = key.bytes().all(|b| b.is_ascii_alphanumeric() || b"_-.".contains(&b));
    let stem = key.split('.').next().unwrap_or("").to_ascii_lowercase();
    let reserved = matches!(stem.as_str(), "con" | "prn" | "aux" | "nul")
        || (stem.len() == 4 && (stem.starts_with("com") || stem.starts_with("lpt")) && stem.as_bytes()[3].is_ascii_digit());
    if key.is_empty() || key.len() > 64 || !plain || key.starts_with('.') || key.contains("..") || reserved {
        return Err(format!("bad save key '{key}': use up to 64 letters, digits, '_', '-' or '.'"));
    }
    Ok(())
}

// ---- encoding ---------------------------------------------------------------

/// The file contents for a value.
pub fn encode(v: &Value) -> Result<String, String> {
    let mut out = String::from(HEADER);
    write_value(&mut out, v, 0)?;
    out.push('\n');
    if out.len() > MAX_BYTES {
        return Err(format!("save data is {} KB; the limit is {} KB", out.len() / 1024, MAX_BYTES / 1024));
    }
    Ok(out)
}

const KEYWORDS: &[&str] = &[
    "and", "break", "do", "else", "elseif", "end", "false", "for", "function", "goto", "if", "in", "local", "nil", "not", "or", "repeat",
    "return", "then", "true", "until", "while",
];

fn is_ident(s: &[u8]) -> bool {
    let ok_start = s.first().is_some_and(|c| c.is_ascii_alphabetic() || *c == b'_');
    ok_start && s.iter().all(|c| c.is_ascii_alphanumeric() || *c == b'_') && !KEYWORDS.iter().any(|k| k.as_bytes() == s)
}

fn write_string(out: &mut String, s: &[u8]) {
    let escape = |out: &mut String, c: char| match c {
        '"' => out.push_str("\\\""),
        '\\' => out.push_str("\\\\"),
        '\n' => out.push_str("\\n"),
        '\r' => out.push_str("\\r"),
        '\t' => out.push_str("\\t"),
        // Decimal escapes are always three digits so a following digit is safe.
        c if c < ' ' || c == '\x7f' => write!(out, "\\{:03}", c as u32).unwrap(),
        c => out.push(c),
    };
    out.push('"');
    match std::str::from_utf8(s) {
        // Valid UTF-8 stays readable; otherwise every non-ASCII byte is escaped.
        Ok(text) => text.chars().for_each(|c| escape(out, c)),
        Err(_) => s.iter().for_each(|&b| if b < 0x80 { escape(out, b as char) } else { write!(out, "\\{b:03}").unwrap() }),
    }
    out.push('"');
}

fn write_number(out: &mut String, n: f64) -> Result<(), String> {
    if !n.is_finite() {
        return Err(format!("can't save the number {n}"));
    }
    // Debug formatting is the shortest text that reads back to the same f64,
    // and always has a '.' or exponent, so it reloads as a float.
    write!(out, "{n:?}").unwrap();
    Ok(())
}

fn write_value(out: &mut String, v: &Value, depth: usize) -> Result<(), String> {
    match v {
        Value::Boolean(b) => out.push_str(if *b { "true" } else { "false" }),
        Value::Integer(i) => write!(out, "{i}").unwrap(),
        Value::Number(n) => write_number(out, *n)?,
        Value::String(s) => write_string(out, &s.as_bytes()),
        Value::Table(t) => write_table(out, t, depth + 1)?,
        Value::Nil => out.push_str("nil"),
        other => return Err(format!("can't save a {}", other.type_name())),
    }
    Ok(())
}

fn key_rank(k: &Value) -> (u8, f64, Vec<u8>) {
    match k {
        Value::Boolean(b) => (0, *b as u8 as f64, Vec::new()),
        Value::Integer(i) => (1, *i as f64, Vec::new()),
        Value::Number(n) => (1, *n, Vec::new()),
        Value::String(s) => (2, 0.0, s.as_bytes().to_vec()),
        _ => (3, 0.0, Vec::new()),
    }
}

fn write_table(out: &mut String, t: &Table, depth: usize) -> Result<(), String> {
    if depth > MAX_DEPTH {
        return Err("save data is nested too deeply (or a table contains itself)".into());
    }
    let mut pairs = Vec::new();
    for p in t.pairs::<Value, Value>() {
        pairs.push(p.map_err(|e| e.to_string())?);
    }
    if pairs.is_empty() {
        out.push_str("{}");
        return Ok(());
    }
    pairs.sort_by(|a, b| {
        let (ra, rb) = (key_rank(&a.0), key_rank(&b.0));
        ra.0.cmp(&rb.0).then(ra.1.total_cmp(&rb.1)).then(ra.2.cmp(&rb.2))
    });
    // A proper sequence (keys exactly 1..n) is written as a plain list.
    let list = pairs.iter().enumerate().all(|(i, (k, _))| matches!(k, Value::Integer(n) if *n == i as i64 + 1));
    let pad = "  ".repeat(depth);
    out.push_str("{\n");
    for (k, v) in &pairs {
        out.push_str(&pad);
        if !list {
            match k {
                Value::String(s) if is_ident(&s.as_bytes()) => out.push_str(&s.to_str().unwrap()),
                Value::String(_) | Value::Integer(_) | Value::Number(_) | Value::Boolean(_) => {
                    out.push('[');
                    write_value(out, k, depth)?;
                    out.push(']');
                }
                other => return Err(format!("can't save a table key of type {}", other.type_name())),
            }
            out.push_str(" = ");
        }
        write_value(out, v, depth)?;
        out.push_str(",\n");
    }
    out.push_str(&"  ".repeat(depth - 1));
    out.push('}');
    Ok(())
}

// ---- decoding ---------------------------------------------------------------

/// Parse a save file back into a Lua value. Accepts what `encode` writes
/// and what a person might reasonably type: comments, single quotes, `;`
/// separators, trailing commas and a leading `return`.
pub fn decode(lua: &Lua, src: &str) -> Result<Value, String> {
    let mut p = Parser { s: src.as_bytes(), i: 0, lua };
    p.skip();
    if p.word() == Some("return") {
        p.i += 6;
    }
    let v = p.value(0)?;
    p.skip();
    if p.i < p.s.len() {
        return Err(p.err("unexpected text after the value"));
    }
    Ok(v)
}

struct Parser<'a> {
    s: &'a [u8],
    i: usize,
    lua: &'a Lua,
}

impl Parser<'_> {
    fn err(&self, what: &str) -> String {
        let line = 1 + self.s[..self.i.min(self.s.len())].iter().filter(|&&c| c == b'\n').count();
        format!("line {line}: {what}")
    }

    fn peek(&self) -> Option<u8> {
        self.s.get(self.i).copied()
    }

    /// Skip whitespace and `--` comments, including `--[[ long ]]` ones.
    fn skip(&mut self) {
        loop {
            while self.peek().is_some_and(|c| c.is_ascii_whitespace()) {
                self.i += 1;
            }
            if !self.s[self.i..].starts_with(b"--") {
                return;
            }
            self.i += 2;
            if self.s[self.i..].starts_with(b"[[") {
                let end = self.s[self.i..].windows(2).position(|w| w == b"]]");
                self.i = end.map_or(self.s.len(), |e| self.i + e + 2);
            } else {
                while self.peek().is_some_and(|c| c != b'\n') {
                    self.i += 1;
                }
            }
        }
    }

    /// The identifier at the cursor, without consuming it.
    fn word(&self) -> Option<&str> {
        let len = self.s[self.i..].iter().take_while(|c| c.is_ascii_alphanumeric() || **c == b'_').count();
        let w = &self.s[self.i..self.i + len];
        is_ident(w).then(|| std::str::from_utf8(w).unwrap()).or_else(|| {
            // Keywords aren't identifiers, but true/false/nil/return are words here.
            let w = std::str::from_utf8(w).ok()?;
            ["true", "false", "nil", "return"].contains(&w).then_some(w)
        })
    }

    fn expect(&mut self, c: u8) -> Result<(), String> {
        self.skip();
        if self.peek() != Some(c) {
            return Err(self.err(&format!("expected '{}'", c as char)));
        }
        self.i += 1;
        Ok(())
    }

    fn value(&mut self, depth: usize) -> Result<Value, String> {
        self.skip();
        match self.peek() {
            Some(b'{') => self.table(depth + 1),
            Some(q @ (b'"' | b'\'')) => self.string(q),
            Some(b'-' | b'.' | b'0'..=b'9') => self.number(),
            Some(_) => {
                let (v, len) = match self.word() {
                    Some("true") => (Value::Boolean(true), 4),
                    Some("false") => (Value::Boolean(false), 5),
                    Some("nil") => (Value::Nil, 3),
                    _ => return Err(self.err("expected a value (table, string, number, true, false or nil)")),
                };
                self.i += len;
                Ok(v)
            }
            None => Err(self.err("unexpected end of file")),
        }
    }

    fn table(&mut self, depth: usize) -> Result<Value, String> {
        if depth > MAX_DEPTH {
            return Err(self.err("tables nested too deeply"));
        }
        self.i += 1; // {
        let t = self.lua.create_table().map_err(|e| e.to_string())?;
        let mut next = 1i64;
        loop {
            self.skip();
            if self.peek() == Some(b'}') {
                self.i += 1;
                return Ok(Value::Table(t));
            }
            let key = if self.peek() == Some(b'[') && self.s.get(self.i + 1) != Some(&b'[') {
                self.i += 1;
                let k = self.value(depth)?;
                self.expect(b']')?;
                self.expect(b'=')?;
                Some(k)
            } else if let Some(w) = self.word().filter(|w| !["true", "false", "nil"].contains(w)) {
                let w = w.to_string();
                self.i += w.len();
                self.expect(b'=')?;
                Some(Value::String(self.lua.create_string(&w).map_err(|e| e.to_string())?))
            } else {
                None
            };
            let v = self.value(depth)?;
            let k = match key {
                Some(Value::Nil) => return Err(self.err("a table key can't be nil")),
                Some(Value::Number(n)) if n.is_nan() => return Err(self.err("a table key can't be NaN")),
                Some(k) => k,
                None => {
                    next += 1;
                    Value::Integer(next - 1)
                }
            };
            t.raw_set(k, v).map_err(|e| e.to_string())?;
            self.skip();
            match self.peek() {
                Some(b',' | b';') => self.i += 1,
                Some(b'}') => {}
                _ => return Err(self.err("expected ',' or '}'")),
            }
        }
    }

    fn string(&mut self, quote: u8) -> Result<Value, String> {
        self.i += 1;
        let mut out = Vec::new();
        loop {
            let Some(c) = self.peek() else { return Err(self.err("unfinished string")) };
            self.i += 1;
            match c {
                c if c == quote => break,
                b'\n' => return Err(self.err("unfinished string")),
                b'\\' => {
                    let Some(e) = self.peek() else { return Err(self.err("unfinished string")) };
                    self.i += 1;
                    match e {
                        b'n' => out.push(b'\n'),
                        b't' => out.push(b'\t'),
                        b'r' => out.push(b'\r'),
                        b'a' => out.push(7),
                        b'b' => out.push(8),
                        b'f' => out.push(12),
                        b'v' => out.push(11),
                        b'\\' | b'"' | b'\'' | b'\n' => out.push(e),
                        b'0'..=b'9' => {
                            let digits = self.s[self.i - 1..].iter().take(3).take_while(|c| c.is_ascii_digit()).count();
                            let n: u32 = std::str::from_utf8(&self.s[self.i - 1..self.i - 1 + digits]).unwrap().parse().unwrap();
                            if n > 255 {
                                return Err(self.err("escape too large"));
                            }
                            out.push(n as u8);
                            self.i += digits - 1;
                        }
                        b'x' => {
                            let hex =
                                self.s.get(self.i..self.i + 2).and_then(|h| u8::from_str_radix(std::str::from_utf8(h).ok()?, 16).ok());
                            out.push(hex.ok_or_else(|| self.err("bad \\x escape"))?);
                            self.i += 2;
                        }
                        b'u' => {
                            let end = self.s[self.i..].iter().position(|&c| c == b'}').ok_or_else(|| self.err("bad \\u escape"))?;
                            let hex = std::str::from_utf8(&self.s[self.i..self.i + end]).unwrap_or("").trim_start_matches('{');
                            let ch =
                                u32::from_str_radix(hex, 16).ok().and_then(char::from_u32).ok_or_else(|| self.err("bad \\u escape"))?;
                            out.extend_from_slice(ch.encode_utf8(&mut [0; 4]).as_bytes());
                            self.i += end + 1;
                        }
                        _ => return Err(self.err("unknown escape")),
                    }
                }
                c => out.push(c),
            }
        }
        Ok(Value::String(self.lua.create_string(&out).map_err(|e| e.to_string())?))
    }

    fn number(&mut self) -> Result<Value, String> {
        let start = self.i;
        if self.peek() == Some(b'-') {
            self.i += 1;
        }
        while let Some(c) = self.peek() {
            let sign_after_e = (c == b'-' || c == b'+') && matches!(self.s[self.i - 1], b'e' | b'E');
            if c.is_ascii_digit() || c == b'.' || c == b'e' || c == b'E' || sign_after_e {
                self.i += 1;
            } else {
                break;
            }
        }
        let text = std::str::from_utf8(&self.s[start..self.i]).unwrap();
        let float = text.contains(['.', 'e', 'E']);
        match (float, text.parse::<i64>(), text.parse::<f64>()) {
            (false, Ok(i), _) => Ok(Value::Integer(i)),
            (_, _, Ok(f)) if f.is_finite() => Ok(Value::Number(f)),
            _ => Err(self.err(&format!("bad number '{text}'"))),
        }
    }
}
```

```rs
//! Just enough linear algebra. Column-major 4x4 matrices, right-handed, Y up.

use std::ops::{Add, AddAssign, Mul, Neg, Sub, SubAssign};

#[derive(Clone, Copy, Debug, Default, PartialEq)]
pub struct Vec3 {
    pub x: f32,
    pub y: f32,
    pub z: f32,
}

pub const fn v3(x: f32, y: f32, z: f32) -> Vec3 {
    Vec3 { x, y, z }
}

impl Vec3 {
    pub const ZERO: Vec3 = v3(0.0, 0.0, 0.0);
    pub const ONE: Vec3 = v3(1.0, 1.0, 1.0);
    pub const UP: Vec3 = v3(0.0, 1.0, 0.0);

    pub fn dot(self, o: Vec3) -> f32 {
        self.x * o.x + self.y * o.y + self.z * o.z
    }
    pub fn cross(self, o: Vec3) -> Vec3 {
        v3(self.y * o.z - self.z * o.y, self.z * o.x - self.x * o.z, self.x * o.y - self.y * o.x)
    }
    pub fn len(self) -> f32 {
        self.dot(self).sqrt()
    }
    pub fn norm(self) -> Vec3 {
        let l = self.len();
        if l > 1e-8 { self * (1.0 / l) } else { Vec3::ZERO }
    }
    pub fn mul(self, o: Vec3) -> Vec3 {
        v3(self.x * o.x, self.y * o.y, self.z * o.z)
    }
    pub fn abs(self) -> Vec3 {
        v3(self.x.abs(), self.y.abs(), self.z.abs())
    }
    pub fn get(self, axis: usize) -> f32 {
        [self.x, self.y, self.z][axis]
    }
    pub fn set(&mut self, axis: usize, v: f32) {
        match axis {
            0 => self.x = v,
            1 => self.y = v,
            _ => self.z = v,
        }
    }
    /// Rotate around +Y by `yaw` radians. Forward at yaw is (sin yaw, 0, cos yaw).
    pub fn rot_y(self, yaw: f32) -> Vec3 {
        let (s, c) = yaw.sin_cos();
        v3(c * self.x + s * self.z, self.y, -s * self.x + c * self.z)
    }
}

impl Add for Vec3 {
    type Output = Vec3;
    fn add(self, o: Vec3) -> Vec3 {
        v3(self.x + o.x, self.y + o.y, self.z + o.z)
    }
}
impl Sub for Vec3 {
    type Output = Vec3;
    fn sub(self, o: Vec3) -> Vec3 {
        v3(self.x - o.x, self.y - o.y, self.z - o.z)
    }

/// Unit quaternion for orientation. (x, y, z) is the vector part.
#[derive(Clone, Copy, Debug, PartialEq)]
pub struct Quat {
    pub x: f32,
    pub y: f32,
    pub z: f32,
    pub w: f32,
}

impl Quat {
    pub const IDENTITY: Quat = Quat { x: 0.0, y: 0.0, z: 0.0, w: 1.0 };

    pub fn axis_angle(axis: Vec3, angle: f32) -> Quat {
        let a = axis.norm();
        let (s, c) = (angle * 0.5).sin_cos();
        Quat { x: a.x * s, y: a.y * s, z: a.z * s, w: c }
    }

    /// Yaw about +Y, then pitch about local +X, then roll about local +Z.
    pub fn euler(yaw: f32, pitch: f32, roll: f32) -> Quat {
        Quat::axis_angle(Vec3::UP, yaw) * Quat::axis_angle(v3(1., 0., 0.), pitch) * Quat::axis_angle(v3(0., 0., 1.), roll)
    }

    /// Normalized blend toward `b` (the short way round): close enough to a
    /// slerp for the small turns between two ticks.
    pub fn nlerp(self, b: Quat, t: f32) -> Quat {
        let s = if self.x * b.x + self.y * b.y + self.z * b.z + self.w * b.w < 0.0 { -1.0 } else { 1.0 };
        Quat {
            x: self.x + (b.x * s - self.x) * t,
            y: self.y + (b.y * s - self.y) * t,
            z: self.z + (b.z * s - self.z) * t,
            w: self.w + (b.w * s - self.w) * t,
        }
        .norm()
    }

    pub fn conj(self) -> Quat {
        Quat { x: -self.x, y: -self.y, z: -self.z, w: self.w }
    }

    pub fn norm(self) -> Quat {
        let l = (self.x * self.x + self.y * self.y + self.z * self.z + self.w * self.w).sqrt();
        if l < 1e-8 {
            return Quat::IDENTITY;
        }
        Quat { x: self.x / l, y: self.y / l, z: self.z / l, w: self.w / l }
    }

    pub fn rotate(self, v: Vec3) -> Vec3 {
        let u = v3(self.x, self.y, self.z);
        let t = u.cross(v) * 2.0;
        v + t * self.w + u.cross(t)
    }

    /// Local axes in world space: (right-handed x, y, z).
    /// (The columns of the rotation matrix, computed directly.)
    pub fn axes(self) -> [Vec3; 3] {
        let Quat { x, y, z, w } = self;
        let (xx, yy, zz) = (x * x, y * y, z * z);
        let (xy, xz, yz, wx, wy, wz) = (x * y, x * z, y * z, w * x, w * y, w * z);
        [
            v3(1. - 2. * (yy + zz), 2. * (xy + wz), 2. * (xz - wy)),
            v3(2. * (xy - wz), 1. - 2. * (xx + zz), 2. * (yz + wx)),
            v3(2. * (xz + wy), 2. * (yz - wx), 1. - 2. * (xx + yy)),
        ]
    }

    /// Advance by angular velocity `w` (world space, rad/s) over `dt`.
    pub fn integrate(self, w: Vec3, dt: f32) -> Quat {
        let dq = Quat { x: w.x, y: w.y, z: w.z, w: 0.0 } * self;
        let h = 0.5 * dt;
        Quat { x: self.x + dq.x * h, y: self.y + dq.y * h, z: self.z + dq.z * h, w: self.w + dq.w * h }.norm()
    }

    /// Heading: the angle of the local +Z axis projected onto the ground.
    pub fn yaw(self) -> f32 {
        let f = self.rotate(v3(0., 0., 1.));
        f.x.atan2(f.z)
    }
}

impl Mul for Quat {
    type Output = Quat;
    fn mul(self, o: Quat) -> Quat {
        Quat {
            w: self.w * o.w - self.x * o.x - self.y * o.y - self.z * o.z,
            x: self.w * o.x + self.x * o.w + self.y * o.z - self.z * o.y,
            y: self.w * o.y - self.x * o.z + self.y * o.w + self.z * o.x,
            z: self.w * o.z + self.x * o.y - self.y * o.x + self.z * o.w,
        }
    }
}

/// Column-major: m[col*4 + row].
#[derive(Clone, Copy, Debug, PartialEq)]
pub struct Mat4(pub [f32; 16]);

impl Mat4 {
    /// translate * rotY(yaw) * scale
    #[cfg(test)]
    pub fn trs(t: Vec3, yaw: f32, s: Vec3) -> Mat4 {
        let (sn, c) = yaw.sin_cos();
        Mat4([c * s.x, 0., -sn * s.x, 0., 0., s.y, 0., 0., sn * s.z, 0., c * s.z, 0., t.x, t.y, t.z, 1.])
    }

    /// translate * rotate(q) * scale
    pub fn trs_q(t: Vec3, q: Quat, s: Vec3) -> Mat4 {
        let [ax, ay, az] = q.axes();
        let (ax, ay, az) = (ax * s.x, ay * s.y, az * s.z);
        Mat4([ax.x, ax.y, ax.z, 0., ay.x, ay.y, ay.z, 0., az.x, az.y, az.z, 0., t.x, t.y, t.z, 1.])
    }

    pub fn perspective(fovy: f32, aspect: f32, near: f32, far: f32) -> Mat4 {
        let f = 1.0 / (fovy * 0.5).tan();
        let nf = 1.0 / (near - far);
        Mat4([f / aspect, 0., 0., 0., 0., f, 0., 0., 0., 0., (far + near) * nf, -1., 0., 0., 2. * far * near * nf, 0.])
    }

    pub fn ortho(l: f32, r: f32, b: f32, t: f32, n: f32, f: f32) -> Mat4 {
        Mat4([
            2. / (r - l),
            0.,
            0.,
            0.,
            0.,
            2. / (t - b),
            0.,
            0.,
            0.,
            0.,
            -2. / (f - n),
            0.,
            -(r + l) / (r - l),
            -(t + b) / (t - b),
            -(f + n) / (f - n),
            1.,
        ])
    }

    pub fn look_at(eye: Vec3, target: Vec3, up: Vec3) -> Mat4 {
        let f = (target - eye).norm();
        let mut s = f.cross(up).norm();
        if s.len() < 1e-6 {
            s = v3(1., 0., 0.);
        }
        let u = s.cross(f);
        Mat4([s.x, u.x, -f.x, 0., s.y, u.y, -f.y, 0., s.z, u.z, -f.z, 0., -s.dot(eye), -u.dot(eye), f.dot(eye), 1.])
    }
}

impl Mul for Mat4 {
    type Output = Mat4;
    fn mul(self, o: Mat4) -> Mat4 {
        let (a, b) = (&self.0, &o.0);
        let mut r = [0.0; 16];
        for c in 0..4 {
            for row in 0..4 {
                r[c * 4 + row] = (0..4).map(|k| a[k * 4 + row] * b[c * 4 + k]).sum();
            }
        }
        Mat4(r)
    }
}
```

```rs
//! CPU-side meshes: flat-shaded triangles with per-vertex color.
//! Built-in unit primitives, Wavefront OBJ (+MTL Kd colors), and baking
//! several parts into one mesh so a prefab is a single draw call.

use crate::math::{Quat, Vec3, v3};
use std::collections::HashMap;
use std::path::Path;

/// 20 bytes: a city is millions of triangles, so normal and colour are bytes.
#[repr(C)]
#[derive(Clone, Copy, Debug)]
pub struct Vertex {
    pub pos: [f32; 3],
    /// Unit normal, each axis -1..1 stored as 0..254 (127 is zero); w unused.
    pub normal: [u8; 4],
    /// Color 0..255. Alpha is emissive: 0 = lit normally, 255 = glows at
    /// night (headlights, windows, bulbs).
    pub color: [u8; 4],
}

fn unorm(x: f32) -> u8 {
    (x.clamp(0.0, 1.0) * 255.0).round() as u8
}

fn snorm(x: f32) -> u8 {
    (x.clamp(-1.0, 1.0) * 127.0).round() as i32 as u8 + 127
}

impl Vertex {
    pub fn new(p: Vec3, n: Vec3, col: Vec3, emissive: f32) -> Vertex {
        Vertex {
            pos: [p.x, p.y, p.z],
            normal: [snorm(n.x), snorm(n.y), snorm(n.z), 0],
            color: [unorm(col.x), unorm(col.y), unorm(col.z), unorm(emissive)],
        }
    }

    pub fn pos(&self) -> Vec3 {
        v3(self.pos[0], self.pos[1], self.pos[2])
    }

    pub fn normal(&self) -> Vec3 {
        let f = |b: u8| (b as f32 - 127.0) / 127.0;
        v3(f(self.normal[0]), f(self.normal[1]), f(self.normal[2]))
    }

    pub fn color(&self) -> Vec3 {
        v3(self.color[0] as f32 / 255.0, self.color[1] as f32 / 255.0, self.color[2] as f32 / 255.0)
    }

    pub fn emissive(&self) -> f32 {
        self.color[3] as f32 / 255.0
    }
}

#[derive(Clone, Debug, Default)]
pub struct Mesh {
    pub verts: Vec<Vertex>,
    pub indices: Vec<u32>,
}

impl Mesh {
    pub fn tri(&mut self, a: Vec3, b: Vec3, c: Vec3, col: Vec3) {
        self.tri_e(a, b, c, col, 0.0);
    }

    pub fn tri_e(&mut self, a: Vec3, b: Vec3, c: Vec3, col: Vec3, emissive: f32) {
        let n = (b - a).cross(c - a).norm();
        let base = self.verts.len() as u32;
        for p in [a, b, c] {
            self.verts.push(Vertex::new(p, n, col, emissive));
        }
        self.indices.extend_from_slice(&[base, base + 1, base + 2]);
    }

    /// Counter-clockwise quad (seen from the front). Flat, so its two
    /// triangles share four vertices.
    pub fn quad(&mut self, a: Vec3, b: Vec3, c: Vec3, d: Vec3, col: Vec3) {
        let n = (b - a).cross(c - a).norm();
        let base = self.verts.len() as u32;
        for p in [a, b, c, d] {
            self.verts.push(Vertex::new(p, n, col, 0.0));
        }
        self.indices.extend_from_slice(&[base, base + 1, base + 2, base, base + 2, base + 3]);
    }

    /// Point the mesh's Y axis along `axis` ("x" or "z"); e.g. a cylinder becomes a wheel.
    pub fn reorient(&mut self, axis: &str) -> Result<(), String> {
        let f: fn([f32; 3]) -> [f32; 3] = match axis {
            "y" => return Ok(()),
            "x" => |[x, y, z]| [y, -x, z], // rotate -90deg about Z
            "z" => |[x, y, z]| [x, -z, y], // rotate +90deg about X
            o => return Err(format!("bad axis '{o}'")),
        };
        for v in &mut self.verts {
            let (p, n) = (f(v.pos), v.normal());
            let n = f([n.x, n.y, n.z]);
            *v = Vertex::new(v3(p[0], p[1], p[2]), v3(n[0], n[1], n[2]), v.color(), v.emissive());
        }
        Ok(())
    }

    pub fn tri_count(&self) -> usize {
        self.indices.len() / 3
    }

    /// Append `src` scaled, turned (yaw, then pitch about its own +X, then
    /// roll about its +Z, as Quat::euler orders them), moved by `offset`,
    /// and tinted by `col`.
    pub fn append_tilted(&mut self, src: &Mesh, offset: Vec3, (yaw, pitch, roll): (f32, f32, f32), scale: Vec3, col: Vec3, emissive: f32) {
        let base = self.verts.len() as u32;
        let inv = v3(1.0 / scale.x, 1.0 / scale.y, 1.0 / scale.z);
        let tilt =
            (pitch != 0.0 || roll != 0.0).then(|| Quat::axis_angle(v3(1.0, 0.0, 0.0), pitch) * Quat::axis_angle(v3(0.0, 0.0, 1.0), roll));
        let turn = |v: Vec3| tilt.map_or(v, |q| q.rotate(v)).rot_y(yaw);
        for v in &src.verts {
            let p = turn(v.pos().mul(scale)) + offset;
            let n = turn(v.normal().mul(inv)).norm();
            self.verts.push(Vertex::new(p, n, v.color().mul(col), v.emissive().max(emissive)));
        }
        self.indices.extend(src.indices.iter().map(|i| i + base));
    }
}

/// Unit cube centered on the origin.
pub fn cube() -> Mesh {
    let mut m = Mesh::default();
    let h = 0.5;
    let p = |x: f32, y: f32, z: f32| v3(x * h, y * h, z * h);
    let w = Vec3::ONE;
    m.quad(p(-1., -1., 1.), p(1., -1., 1.), p(1., 1., 1.), p(-1., 1., 1.), w); // +z
    m.quad(p(1., -1., -1.), p(-1., -1., -1.), p(-1., 1., -1.), p(1., 1., -1.), w); // -z
    m.quad(p(1., -1., 1.), p(1., -1., -1.), p(1., 1., -1.), p(1., 1., 1.), w); // +x
    m.quad(p(-1., -1., -1.), p(-1., -1., 1.), p(-1., 1., 1.), p(-1., 1., -1.), w); // -x
    m.quad(p(-1., 1., 1.), p(1., 1., 1.), p(1., 1., -1.), p(-1., 1., -1.), w); // +y
    m.quad(p(-1., -1., -1.), p(1., -1., -1.), p(1., -1., 1.), p(-1., -1., 1.), w); // -y
    m
}

/// Unit ramp: full height at +z, zero height at -z.
pub fn wedge() -> Mesh {
    let mut m = Mesh::default();
    let w = Vec3::ONE;
    let (a, b) = (v3(-0.5, -0.5, -0.5), v3(0.5, -0.5, -0.5));
    let (c, d) = (v3(0.5, -0.5, 0.5), v3(-0.5, -0.5, 0.5));
    let (e, f) = (v3(0.5, 0.5, 0.5), v3(-0.5, 0.5, 0.5));
    m.quad(a, b, c, d, w); // bottom
    m.quad(d, c, e, f, w); // back wall +z
    m.quad(a, f, e, b, w); // slope
    m.tri(a, d, f, w); // -x side
    m.tri(b, e, c, w); // +x side
    m
}

/// Unit cylinder along Y (diameter 1, height 1).
pub fn cylinder(sides: usize) -> Mesh {
    let mut m = Mesh::default();
    let w = Vec3::ONE;
    let pt = |i: usize, y: f32| {
        let a = i as f32 / sides as f32 * std::f32::consts::TAU;
        v3(a.cos() * 0.5, y, -a.sin() * 0.5)
    };
    for i in 0..sides {
        let (a0, a1) = (pt(i, -0.5), pt(i + 1, -0.5));
        let (b0, b1) = (pt(i, 0.5), pt(i + 1, 0.5));
        m.quad(a0, a1, b1, b0, w);
        m.tri(v3(0., 0.5, 0.), b0, b1, w);
        m.tri(v3(0., -0.5, 0.), a1, a0, w);
    }
    m
}

/// Low-poly UV sphere, diameter 1.
pub fn sphere(rings: usize, segs: usize) -> Mesh {
    let mut m = Mesh::default();
    let pt = |r: usize, s: usize| {
        let th = r as f32 / rings as f32 * std::f32::consts::PI;
        let ph = s as f32 / segs as f32 * std::f32::consts::TAU;
        v3(th.sin() * ph.cos() * 0.5, th.cos() * 0.5, -th.sin() * ph.sin() * 0.5)
    };
    for r in 0..rings {
        for s in 0..segs {
            let (a, b, c, d) = (pt(r, s), pt(r + 1, s), pt(r + 1, s + 1), pt(r, s + 1));
            if r != 0 {
                m.tri(a, b, d, Vec3::ONE);
            }
            if r != rings - 1 {
                m.tri(b, c, d, Vec3::ONE);
            }
        }
    }
    m
}

/// Unit quad in XZ facing +Y.
pub fn plane() -> Mesh {
    let mut m = Mesh::default();
    m.quad(v3(-0.5, 0., 0.5), v3(0.5, 0., 0.5), v3(0.5, 0., -0.5), v3(-0.5, 0., -0.5), Vec3::ONE);
    m
}

    BUILTIN_NAMES.contains(&name)

/// Minimal Wavefront OBJ: v, f (any polygon, fan-triangulated, v/vt/vn and negative
/// indices), mtllib + usemtl for flat Kd colors. Normals are recomputed flat.
pub fn load_obj(path: &Path) -> Result<Mesh, String> {
    let src = std::fs::read_to_string(path).map_err(|e| format!("{}: {e}", path.display()))?;
    let dir = path.parent().unwrap_or(Path::new("."));
    let mut pos: Vec<Vec3> = Vec::new();
    let mut mats: HashMap<String, (Vec3, f32)> = HashMap::new();
    let (mut col, mut glow) = (Vec3::ONE, 0.0);
    let mut m = Mesh::default();
    let num = |s: Option<&str>| s.and_then(|s| s.parse::<f32>().ok()).unwrap_or(0.0);
    for (ln, line) in src.lines().enumerate() {
        let mut it = line.split_whitespace();
        match it.next() {
            Some("v") => pos.push(v3(num(it.next()), num(it.next()), num(it.next()))),
            Some("mtllib") => {
                if let Some(f) = it.next()
                    && let Ok(txt) = std::fs::read_to_string(dir.join(f))
                {
                    parse_mtl(&txt, &mut mats);
                }
            }
            Some("usemtl") => (col, glow) = it.next().and_then(|n| mats.get(n).copied()).unwrap_or((Vec3::ONE, 0.0)),
            Some("f") => {
                let idx: Result<Vec<Vec3>, String> = it
                    .map(|tok| {
                        let i: i64 = tok.split('/').next().unwrap_or("").parse().map_err(|_| format!("line {}: bad face", ln + 1))?;
                        let i = if i < 0 { pos.len() as i64 + i } else { i - 1 };
                        pos.get(i as usize).copied().ok_or(format!("line {}: index out of range", ln + 1))
                    })
                    .collect();
                let p = idx?;
                for k in 1..p.len().saturating_sub(1) {
                    m.tri_e(p[0], p[k], p[k + 1], col, glow);
                }
            }
            _ => {}
        }
    }
    Ok(m)
}

/// Kd is the color; Ke (emission) marks the material as glowing at night.
fn parse_mtl(txt: &str, out: &mut HashMap<String, (Vec3, f32)>) {
    let mut cur = String::new();
    for line in txt.lines() {
        let mut it = line.split_whitespace();
        let key = it.next();
        let c: Vec<f32> = it.filter_map(|s| s.parse().ok()).collect();
        match key {
            Some("newmtl") => {
                cur = line.split_whitespace().nth(1).unwrap_or("").to_string();
                out.insert(cur.clone(), (Vec3::ONE, 0.0));
            }
            Some("Kd") if c.len() >= 3 => out.entry(cur.clone()).or_insert((Vec3::ONE, 0.0)).0 = v3(c[0], c[1], c[2]),
            Some("Ke") if c.len() >= 3 => out.entry(cur.clone()).or_insert((Vec3::ONE, 0.0)).1 = c[0].max(c[1]).max(c[2]).min(1.0),
            _ => {}
        }
    }
}

        let v = Vertex::new(v3(1.5, -2.0, 3.25), n, v3(0.2, 0.55, 1.0), 0.6);
        assert_eq!(v.pos(), v3(1.5, -2.0, 3.25));
        assert!((v.normal() - n).len() < 0.01, "{:?}", v.normal());
        assert!((v.color() - v3(0.2, 0.55, 1.0)).len() < 0.005, "{:?}", v.color());
        assert!((v.emissive() - 0.6).abs() < 0.005);
        // Out of range clamps rather than wrapping round.
        let w = Vertex::new(Vec3::ZERO, v3(0.0, 1.0, 0.0), v3(1.4, -0.2, 0.5), 2.0);
        assert_eq!((w.color[0], w.color[1], w.color[3]), (255, 0, 255));
    }

    #[test]
    fn tilted_parts_climb_where_they_point() {
        // A 10 m plank turned to face +X and pitched up 0.2 rad: its far end
        // is 10 sin(0.2) higher than its near end, and it's still outward-facing.
        let mut m = Mesh::default();
        m.append_tilted(&cube(), Vec3::ZERO, (std::f32::consts::FRAC_PI_2, -0.2, 0.0), v3(1.0, 0.1, 10.0), Vec3::ONE, 0.0);
        let (lo, hi) = m.verts.iter().fold((v3(9., 9., 9.), v3(-9., -9., -9.)), |(lo, hi), v| {
            let p = v3(v.pos[0], v.pos[1], v.pos[2]);
            (if p.x < lo.x { p } else { lo }, if p.x > hi.x { p } else { hi })
        });
        assert!(hi.x > 4.8 && lo.x < -4.8, "{lo:?} {hi:?}");
        assert!((hi.y - lo.y - 10.0 * 0.2f32.sin()).abs() < 0.15, "rise {}", hi.y - lo.y);
        outward(&m);
    }
```
